async function ask(q: string): Promise<string> {
  let a: string | null = null;
  for (let i = 0; i < 3 && a !== "y" && a !== "n"; i++) {
    try {
      a = (await rl.question(`${q} (y/n) `, { signal: AbortSignal.timeout(10 * 60_000) })).trim().toLowerCase();
    } catch {
      a = null; // no answer in 10 minutes, or stdin closed
      break;
    }
  }
  const answer = a === "y" || a === "n" ? a : "no answer";
  result.answers.push({ q, answer, at: at() });
  transcript.push(`    ${q} -> ${answer}`);
  if (answer !== "y") result.failures.push(`answered ${answer}: ${q}`);
  save();
  return answer;
}

type Called = { ok: true; ms: number; json: Record<string, any> } | { ok: false; ms: number; error: unknown };
/** A tool call that gives up after STEP_MS: its JSON, or its error body. */
async function call(label: string, fn: () => Promise<{ json: Record<string, any> }>): Promise<Called> {
  const s = performance.now();
  let timer: NodeJS.Timeout | undefined;
  const late = new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`${label}: no answer within ${STEP_MS / 1000} s`)), STEP_MS);
  });
  try {
    const out = await Promise.race([fn(), late]);
    return { ok: true, ms: Math.round(performance.now() - s), json: out.json };
  } catch (err) {
    return { ok: false, ms: Math.round(performance.now() - s), error: toToolError(err).body() };
  } finally {
    clearTimeout(timer);
  }
}

  check(set.ok && set.json["failed"].length === 0 && near(g?.after, SPOT), set.ok ? `the GPS position was written and read back: ${JSON.stringify(g?.after ?? null)} for ${JSON.stringify(SPOT)}, ${ms(set)}` : `the GPS position was written (${why(set)})`);
  say(`NOW: select ${FILENAME} in the Library module and look at the Metadata panel (right side), field GPS.`);
  await ask(`Does GPS read about ${SPOT_DMS} (the seconds may show decimals)?`);
} catch (err: any) {
  result.error = String(err?.message ?? err);
  result.failures.push(`error: ${result.error}`);
  say(`ERROR: ${result.error}`);
} finally {
  // 5. The put-back, whatever happened.
  try {
    if (keywordTried && !hasKeyword(keywordsBefore)) {
      const rm = await call("lr_set_keywords", () => tools.setKeywords({ uuids: [UUID], remove: [KEYWORD] }));
      const r = entry(rm);
      const after: unknown = r?.after;
      const back =
        rm.ok && rm.json["failed"].length === 0 && Array.isArray(after) && (keywordsBefore ? sameSet(after, keywordsBefore) : !hasKeyword(after));
      step("5_keyword_back", { ms: rm.ms, ok: rm.ok, after_count: Array.isArray(after) ? after.length : null, failed: rm.ok ? rm.json["failed"] : rm.error });
      check(rm.ok && back, back ? "PUT BACK keywords: YES (the photo's keywords as before the check)" : `PUT BACK keywords: NO (${why(rm)})`);
      if (!back) result.cannot_put_back.push(`the keyword "${KEYWORD}" on ${FILENAME}: take it off in the Keyword List panel (untick it with the photo selected)`);
    }
    if (gpsMayBeWritten && gpsBefore === undefined) {
      check(false, "PUT BACK GPS: NO (the write may have landed, and the position before it was never read)");
      result.cannot_put_back.push(`the GPS position of ${FILENAME}: its earlier position is unknown`);
    } else if (gpsMayBeWritten) {
      const back = await call("lr_set_gps", () => tools.setGps({ uuids: [UUID], position: gpsBefore }));
      const r = entry(back);
      const ok = back.ok && back.json["failed"].length === 0 && (gpsBefore === null ? r?.after === null : near(r?.after, gpsBefore));
      step("5_gps_back", { ms: back.ms, ok: back.ok, removed: gpsBefore === null, put_back: ok, wrote: r?.changed ?? null, failed: back.ok ? back.json["failed"] : back.error });
      check(ok, ok ? `PUT BACK GPS: YES (${gpsBefore === null ? "removed again with position null: the photo had none" : "the photo's own position again"})` : `PUT BACK GPS: NO (${why(back)})`);
      if (!ok) result.cannot_put_back.push(`the GPS position of ${FILENAME}: ${gpsBefore === null ? "Lightroom did not remove it; delete the GPS field's text in the Metadata panel" : "set it back in the Metadata panel"}`);
      say("NOW: look at the Metadata panel's GPS field again.");
      await ask(gpsBefore === null ? "Is the GPS field empty again, as before the check?" : "Does the GPS field show the photo's own position again, as before the check?");
      if (back.ok && r?.changed === true) {
        say("NOW: click Lightroom's Edit menu (top menu bar) and read its first item. Do not click it; press Esc to close the menu.");
        await ask(`Does the first item read "Undo AVG set GPS"?`);
      } else {
        say("(The put-back wrote no GPS position, so the Edit menu is not checked.)");
      }
    } else if (gpsTried) {
      say("GPS: the set was refused before anything was written, so there is nothing to put back.");
    }
    if (start) {
      const end = await settingsNow();
      const diff = [...new Set([...Object.keys(start), ...Object.keys(end)])].filter((k) => JSON.stringify(start?.[k]) !== JSON.stringify(end[k]));
      step("5_develop", { differing: diff });
      check(diff.length === 0, diff.length === 0 ? "no Develop setting changed" : `Develop settings changed: ${diff.join(", ")}`);
    }
  } catch (err: any) {
    check(false, "PUT BACK ran", String(err?.message ?? err));
  }
  const created = [PARENT, KEYWORD].filter((k) => existedBefore === null || !existedBefore.includes(k));
  if (keywordTried && created.length > 0) {
    result.cannot_put_back.push(
      `the keywords ${created.map((k) => `"${k}"`).join(" and ")} stay in the catalog's Keyword List: the SDK lists no call that deletes a keyword ` +
        `[handle: https://lrc.mcor.dev/modules/LrCatalog.html, https://lrc.mcor.dev/modules/LrKeyword.html]. To remove them: Keyword List panel, right-click "${PARENT}" > Delete`,
    );
  }

  // 6. The bind-failure log line.
  if (existsSync(bridgeLog)) {
    const lines = readFileSync(bridgeLog, "utf8").split(/\r?\n/);
    const from = lines.reduce((last, line, i) => (line.includes("bridge: starting generation") ? i : last), 0);
    const failed = lines.slice(from).filter((line) => line.includes("failed to open"));
    result.bridge_log = { file: bridgeLog, since: lines[from] ?? null, failed_to_open: failed.length, last: failed.slice(-3) };
    say(`bridge.log since the plugin last started: ${failed.length} "failed to open" line(s)${failed.length ? `; last: ${failed.at(-1)}` : ""}`);
  } else {
    result.bridge_log = { file: bridgeLog, found: false };
    say(`bridge.log not found at ${bridgeLog}`);
  }

  result.bridge = client.stats;
  result.ended = at();
  client.stop();
  await lock.lock.release();
  rl.close();
  const failures: string[] = result.failures;
  say("");
  say(failures.length === 0 ? "automaat-sync check: WORKED" : `automaat-sync check: FAILED (${failures.length})`);
  for (const f of failures) say(`  - ${f}`);
  if (result.cannot_put_back.length > 0) {
    say("NOT PUT BACK (by design or by failure):");
    for (const c of result.cannot_put_back) say(`  - ${c}`);
  }
  say(`Results: ${path.join(outDir, "check.json")} and check.txt`);
  process.exit(failures.length === 0 ? 0 : 1);
}
