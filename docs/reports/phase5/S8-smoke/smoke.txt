##### Appendix: the smoke kit, verbatim

=== fake-lr.lua
-- A fake Lightroom SDK for smoke-running plugin\spikes\S8.lrplugin under fengari (Lua 5.3).
-- Tasks are coroutines on a virtual clock; files live in memory (fengari has no io.open);
-- views are plain tables; dialogs are driven by the scenario script.

FAKE = { now = 1000, tasks = {}, files = {}, messages = {}, hud = nil, modal_answers = true,
    close_exists = true, honour_enabled = true, target = true, log = {} }

local function flog(s) table.insert(FAKE.log, string.format("[%7.2f] %s", FAKE.now - 1000, s)) end
FAKE.flog = flog

-- io shim --------------------------------------------------------------------------------------
local function fileHandle(path, mode)
    local h = { path = path, mode = mode, buf = {} }
    if mode:find("a") then h.buf[1] = FAKE.files[path] or "" end
    function h:write(...)
        for _, s in ipairs({ ... }) do table.insert(self.buf, tostring(s)) end
        return self
    end
    function h:read(fmt) local t = FAKE.files[self.path]; self.consumed = true; return t end
    function h:lines()
        local t = FAKE.files[self.path] or ""
        return t:gmatch("([^\n]*)\n?")
    end
    function h:close()
        if self.mode:find("[wa]") then FAKE.files[self.path] = table.concat(self.buf) end
        return true
    end
    return h
end
io = io or {}
io.open = function(path, mode)
    mode = mode or "r"
    if mode:find("r") and FAKE.files[path] == nil then return nil, path .. ": No such file" end
    return fileHandle(path, mode)
end

-- scheduler ------------------------------------------------------------------------------------
local nextSeq
local function spawn(fn, name)
    local co = coroutine.create(fn)
    table.insert(FAKE.tasks, { co = co, wake = FAKE.now, seq = nextSeq(), name = name or "task" })
end
FAKE.spawn = spawn

local seq = 0
nextSeq = function() seq = seq + 1; return seq end

function FAKE.run(untilTime)
    while true do
        for i = #FAKE.tasks, 1, -1 do
            if coroutine.status(FAKE.tasks[i].co) == "dead" then table.remove(FAKE.tasks, i) end
        end
        local best
        for i, t in ipairs(FAKE.tasks) do
            local b = best and FAKE.tasks[best]
            if not b or t.wake < b.wake or (t.wake == b.wake and t.seq < b.seq) then best = i end
        end
        if not best or FAKE.tasks[best].wake > untilTime then
            if untilTime > FAKE.now then FAKE.now = untilTime end
            return
        end
        local t = FAKE.tasks[best]
        if t.wake > FAKE.now then FAKE.now = t.wake end
        local ok, err = coroutine.resume(t.co)
        if not ok then flog("TASK ERROR in " .. t.name .. ": " .. tostring(err)); error("task error in " .. t.name .. ": " .. tostring(err)) end
    end
end

local function sleep(s)
    local me
    for _, t in ipairs(FAKE.tasks) do if t.co == coroutine.running() then me = t end end
    if not me then error("sleep outside a task") end
    me.wake = FAKE.now + s
    me.seq = nextSeq()
    coroutine.yield()
end

-- SDK modules ----------------------------------------------------------------------------------
local M = {}
M.LrTasks = {
    startAsyncTask = function(fn, name) spawn(fn, name) end,
    sleep = sleep,
    yield = function() sleep(0) end,
    pcall = pcall,
    canYield = function() return coroutine.isyieldable() end,
}
M.LrFunctionContext = {
    postAsyncTaskWithContext = function(name, fn) spawn(function() fn({ name = name }) end, name) end,
}
M.LrDate = {
    currentTime = function() return FAKE.now end,
    timeToUserFormat = function(t, fmt) return string.format("T%.2f", t - 1000) end,
}
M.LrPathUtils = {
    child = function(a, b) return a .. "\\" .. b end,
    getStandardFilePath = function(k) return "%USERPROFILE%\\AppData\\Local\\Temp" end,
    parent = function(p) return p:match("^(.*)\\") end,
}
M.LrFileUtils = { createAllDirectories = function() return true end }
local uuidN = 0
M.LrUUID = { generateUUID = function() uuidN = uuidN + 1; return string.format("UUID-%04d", uuidN) end }

function M_makeProps()
    local data, observers = {}, {}
    return setmetatable({}, {
        __index = function(_, k)
            if k == "addObserver" then return function(_, key, fn) observers[key] = observers[key] or {}; table.insert(observers[key], fn) end end
            return data[k]
        end,
        __newindex = function(self, k, v)
            data[k] = v
            for _, fn in ipairs(observers[k] or {}) do fn(self, k, v) end
        end,
    })
end

function FAKE.click(title)
    local props = FAKE.hud and FAKE.hud.props
    for _, b in ipairs(FAKE.hudButtons or {}) do
        local t = b.spec.title
        if type(t) == "table" then t = props[t.bind] end
        if t == title or (title == "Approve" and type(b.spec.title) == "table") then
            local en = b.spec.enabled
            if type(en) == "table" and FAKE.honour_enabled and not props[en.bind] then
                flog("CLICK " .. title .. " ignored (disabled)")
                return
            end
            flog("CLICK " .. title)
            b.spec.action()
            return
        end
    end
    error("no button " .. title)
end

=== run-smoke.cjs
// Runs main.lua under fengari for one scenario; writes the fake's files to out/<scenario>/ and
// prints the checks. Usage: node run-smoke.cjs <pluginDir> <scenario>
const fs = require("fs");
const path = require("path");
const { lua, lauxlib, lualib, to_luastring } = require("fengari");

const [pluginDir, scenario] = process.argv.slice(2);
const here = __dirname.replaceAll("\\", "/");
const L = lauxlib.luaL_newstate();
lualib.luaL_openlibs(L);
const prelude = `PLUGIN_DIR = ${JSON.stringify(pluginDir.replaceAll("\\", "/"))}; FAKE_DIR = ${JSON.stringify(here)}; SCENARIO = ${JSON.stringify(scenario)}`;
const code = prelude + "\n" + fs.readFileSync(path.join(__dirname, "main.lua"), "utf8");
let out = "";
const origLog = console.log;
const origWrite = process.stdout.write.bind(process.stdout);
process.stdout.write = (s) => { out += s; return true; };
let status = lauxlib.luaL_loadbuffer(L, to_luastring(code), null, to_luastring("main"));
if (status === lua.LUA_OK) status = lua.lua_pcall(L, 0, 0, 0);
process.stdout.write = origWrite;
if (status !== lua.LUA_OK) {
  origLog(out);
  origLog("LUA ERROR: " + lua.lua_tojsstring(L, -1));
  process.exit(2);
}
const outDir = path.join(__dirname, "out", scenario);
fs.rmSync(outDir, { recursive: true, force: true });
fs.mkdirSync(outDir, { recursive: true });
const parts = out.split(/^===/m);
const report = [];
for (const p of parts) {
  if (p.startsWith("FILE ")) {
    const nl = p.indexOf("\n");
    fs.writeFileSync(path.join(outDir, p.slice(5, nl).trim()), p.slice(nl + 1).replace(/\n$/, ""));
  } else report.push(p.startsWith("LOG") || p.startsWith("RESULT") ? "===" + p : p);
}
origLog(report.join(""));


=== parse.cjs
// Parse every .lua under the given folders as Lua 5.1 (syntax only).
const fs = require("fs");
const path = require("path");
const luaparse = require("luaparse");
let ok = 0, bad = 0;
function walk(dir) {
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) walk(p);
    else if (p.endsWith(".lua")) {
      try { luaparse.parse(fs.readFileSync(p, "utf8"), { luaVersion: "5.1" }); ok++; }
      catch (err) { bad++; console.log("FAIL", p, err.message); }
    }
  }
}
for (const d of process.argv.slice(2)) walk(d);
console.log(`${ok}/${ok + bad} Lua files parse as Lua 5.1`);
process.exit(bad ? 1 : 0);
