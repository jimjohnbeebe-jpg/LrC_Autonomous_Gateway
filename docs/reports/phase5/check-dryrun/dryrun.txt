    appendFileSync(bridgeLog, `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}.${p(d.getMilliseconds(), 3)} INFO  ${text}\n`);
  };
  const hudUpdate = plugin.handlers.get("hud_update") as (p: Record<string, unknown>, id: string) => Promise<FakeReply>;
  plugin.handlers.set("hud_update", async (p, id) => {
    const r = await hudUpdate(p, id);
    if (typeof r === "object" && r.ok) {
      const res = r.payload as { applied?: boolean; opened?: boolean };
      if (res.opened) log("hud: shown");
      if (res.applied && ["accepted", "aborted", "ended"].includes(String(p["stage"]))) {
        setTimeout(() => {
          if (!lr.hud.shown) return;
          lr.hud.shown = false;
          log("hud: closed");
        }, 5000);
      }
    }
    return r;
  });
  const click = (name: "hud_abort" | "hud_accept" | "hud_pick" | "hud_approve_pass", extra: Record<string, unknown> = {}, source = "hud"): void => {
    const last = lr.hud.last();
    if (!last) return;
    const id = `dry-${Math.random().toString(36).slice(2, 10)}`;
    log(`hud: ${name} ${id} from the ${source}: sent`);
    hudEvent(plugin, name, { session_id: last.session_id, seq_seen: last.seq, source, ...extra }, id);
  };
  const env = {
    ...process.env,
    LRC_AVG_COMMAND_PORT: String(plugin.commandPort),
    LRC_AVG_EVENT_PORT: String(plugin.eventPort),
    LRC_AVG_TOKEN_FILE: tokenFile,
    LRC_AVG_LOG_DIR: path.join(scratch, "logs"),
    LRC_AVG_P5_OUT: path.join(scratch, "out"),
    LRC_AVG_PLUGIN_LOG: bridgeLog,
    LOCALAPPDATA: path.join(scratch, "localappdata"),
  };

  // "Claude Desktop": one engine for every chat.
  let desk: Client | null = null;
  const notes: string[] = [];
  const call = async (name: string, args: Record<string, unknown>): Promise<Record<string, unknown>> => {
    if (!desk) {
      desk = new Client({ name: "dry-desktop", version: "0" });
      await desk.connect(new StdioClientTransport({ command: process.execPath, args: [path.join(repo, "engine/dist/mcp/main.js")], env: env as Record<string, string>, stderr: "pipe" }));
    }
    const r = (await desk.callTool({ name, arguments: args }, undefined, { timeout: 180000 })) as { isError?: boolean; content: Array<{ type: string; text?: string }> };
    return JSON.parse(r.content.find((x) => x.type === "text")?.text ?? "{}") as Record<string, unknown>;
  };
  const goldenChat = async (): Promise<string> => {
    const begin = await call("lr_begin_session", { intent_id: "landscape_golden_hour", return_image: "none" });
    const sid = String(begin["session_id"]);
    for (let i = 0; i < 2; i++) await call("lr_step", { session_id: sid, settings: { vibrance: 3 }, rationale: "dry run", return_image: "none" });
    const end = await call("lr_end_session", { session_id: sid, outcome: "accept" });
    return `golden chat on ${lr.filename}: ${String(begin["ok"])}, end ${String(end["outcome"])}`;
  };
  const approveChat = async (): Promise<string> => {
    const begin = await call("lr_begin_session", { intent_id: "landscape_golden_hour", return_image: "none" });
    const sid = String(begin["session_id"]);
    const step = () => call("lr_step", { session_id: sid, settings: { vibrance: 3 }, rationale: "dry run", return_image: "none" });
    await step();
    const t0 = Date.now();
    const waited = await step(); // pass 2 waits the full 60 s in vain
    const heldMs = Date.now() - t0;
    click("hud_approve_pass", { pass: 1 });
    await new Promise((r) => setTimeout(r, 500));
    const p2 = await step();
    setTimeout(() => click("hud_approve_pass", { pass: 2 }), 1500);
    const p3 = await step();
    const end = await call("lr_end_session", { session_id: sid, outcome: "accept" });
    const code = (waited["error"] as { code?: unknown } | undefined)?.code; // server.ts: an error is {ok: false, error: {code, …}}
    return `approve chat: first wait answered ${String(code)} after ${heldMs} ms at the MCP client (timeout 180 s); then ${String(p2["pass"])}, ${String(p3["pass"])}; end ${String(end["outcome"])}`;
  };

  const cli = spawn(process.execPath, [path.join(repo, "engine/dist/devtools/phase5-check-cli.js")], { env, stdio: ["pipe", "pipe", "pipe"] });
  let out = "";
  let tail = "";
  let chat = "";
  cli.stderr.on("data", (d) => (out += String(d)));
  const enter = (s = ""): void => void cli.stdin.write(`${s}\n`);
  const later = (ms: number, fn: () => void): void => void setTimeout(fn, ms);
  const says = (line: string): void => {
    const c = /^(The approve chat|Chat \d of 6)/.exec(line);
    if (c) chat = c[1] as string;
    if (/Now a visit to Plug-in Manager/.test(line)) {
      plugin.answerPings = false;
      later(8000, () => (plugin.answerPings = true));
    }
    const mode = /set Mode to "(Approve each pass|Autonomous)"/.exec(line);
    if (mode && lr.prefs) lr.prefs["mode"] = mode[1] === "Autonomous" ? "autonomous" : "approve_each_pass";
    if (/^ {2}Now click Abort in the HUD\./.test(line)) later(500, () => click("hud_abort"));
    if (/then click "Approve pass 1" in the HUD/.test(line)) later(1500, () => click("hud_approve_pass", { pass: 1 }));
    if (/This time click Abort in the HUD/.test(line)) later(1000, () => click("hud_abort"));
    if (/"Pick A", "Pick B" and "Pick C" are now on/.test(line)) later(500, () => click("hud_pick", { variant: "B" }));
    if (/^ {2}Now click Accept in the HUD\./.test(line)) later(500, () => click("hud_accept"));
    if (line.includes(MENU.accept)) later(500, () => click("hud_accept", {}, "menu"));
    if (line.includes(MENU.abort)) later(500, () => click("hud_abort", {}, "menu"));
    if (line.includes(MENU.hud)) later(500, () => ((lr.hud.shown = true), log("hud: shown")));
  };
  let pending = Promise.resolve();
  const prompt = async (text: string): Promise<boolean> => {
    const photo = /[Cc]lick (.+?) \(the original/.exec(text);
    if (photo) {
      lr.selected = lr.uuid;
      lr.filename = photo[1] as string;
      enter();
    } else if (/once more\? Type y or n/.test(text)) enter("n");
    else if (/Type y or n, then Enter:\s*$/.test(text)) enter("y");
    else if (/Press D to go back.*Enter here\.\s*$|Still in the catalog/.test(text)) {
      lr.copies.clear();
      enter();
    } else if (/Come back to this window and press Enter\.\s*$/.test(text)) {
      notes.push(await (chat === "The approve chat" ? approveChat() : goldenChat()));
      enter();
    } else if (/[Pp]ress Enter[^\n]*\.\s*$/.test(text)) enter(); // every other prompt: "Look at it, then press Enter.", "… press Enter here."
    else return false;
    return true;
  };
  cli.stdout.on("data", (d) => {
    const text = String(d);
    out += text;
    pending = pending.then(async () => {
      const lines = (tail + text).split("\n");
      tail = lines.pop() ?? "";
      for (const l of lines) says(l);
      if (tail && (await prompt(tail))) tail = "";
    });
  });
  const code = await new Promise((r) => cli.on("close", r));
  await (desk as Client | null)?.close();
  await plugin.close();
  const transcript = out.split(os.homedir()).join("%USERPROFILE%");
  writeFileSync(path.join(repo, "logs", "scratch-p5-dryrun-output.txt"), `${transcript}\n[driver] ${notes.join("\n[driver] ")}\n[driver] exit code ${String(code)}; History steps ${lr.history.length}; exports ${lr.exports}; copies left ${lr.copies.size}; scratch ${scratch.split(os.homedir()).join("%USERPROFILE%")}\n`);
  console.log(`[driver] exit ${String(code)}`);
  expect(code).toBe(0);
});
