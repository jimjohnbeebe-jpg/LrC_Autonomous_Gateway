== Appendix: the driver (a scratch vitest file, engine\tests\zz-scratch-p4-dryrun.test.ts, run with `npx vitest run tests/zz-scratch-p4-dryrun.test.ts` in engine\, then deleted) ==
// SCRATCH (not committed): dry run of the built `phase4-check-cli.js` against the simulated plugin on
// OS-chosen ports with its own token file, so Jim's Lightroom is not touched; the real
// `engine\dist\mcp\main.js` plays Claude Desktop in Part 2 (a Variants session over stdio MCP).
// LRC_AVG_PRESET_DIR and LRC_AVG_LOG_DIR point at a scratch folder. The CLI's own results go to
// %TEMP%\LrC-AVG\P4 (deleted afterwards by hand).

import { spawn } from "node:child_process";
import { copyFileSync, mkdirSync, mkdtempSync, readdirSync, readFileSync, unlinkSync, writeFileSync } from "node:fs";
import os from "node:os";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { Client } from "@modelcontextprotocol/sdk/client/index.js";
import { StdioClientTransport } from "@modelcontextprotocol/sdk/client/stdio.js";
import { expect, it } from "vitest";
import { parseXml, presetDescription } from "../src/presets/index.js";
import { defaultPreviewDir } from "../src/preview/index.js";
import { FakePlugin } from "./helpers/fake-plugin.js";
import { LightroomSim } from "./helpers/lightroom-sim.js";

const repo = fileURLToPath(new URL("../../", import.meta.url));

it("dry run of npm run phase4:check", { timeout: 300000 }, async () => {
  const scratch = mkdtempSync(path.join(os.tmpdir(), "lrc-avg-p4dry-"));
  const presets = path.join(scratch, "presets");
  const logs = path.join(scratch, "logs");
  mkdirSync(presets);
  mkdirSync(logs);
  copyFileSync(path.join(repo, "engine/tests/fixtures/presets/reference.lrc15.xmp"), path.join(presets, "AVG preset reference.xmp"));
  copyFileSync(path.join(repo, "engine/tests/fixtures/presets/reference-2.lrc15.xmp"), path.join(presets, "AVG preset reference 2.xmp"));
  const plugin = await FakePlugin.start();
  plugin.token = "dry-token-p4";
  const tokenFile = path.join(scratch, "dry_token");
  writeFileSync(tokenFile, plugin.token);
  const lr = new LightroomSim(defaultPreviewDir());
  lr.renderModel = "tonal";
  lr.filename = "20260907-_OZ80099.NEF";
  lr.customWhiteBalanceOnTemperature = true;
  lr.install(plugin);
  const env = { ...process.env, LRC_AVG_COMMAND_PORT: String(plugin.commandPort), LRC_AVG_EVENT_PORT: String(plugin.eventPort), LRC_AVG_TOKEN_FILE: tokenFile, LRC_AVG_PRESET_DIR: presets, LRC_AVG_LOG_DIR: logs };

  const chat = async (): Promise<string> => {
    const t = new StdioClientTransport({ command: process.execPath, args: [path.join(repo, "engine/dist/mcp/main.js")], env: env as Record<string, string>, stderr: "pipe" });
    const c = new Client({ name: "dry-desktop", version: "0" });
    await c.connect(t);
    const call = async (name: string, args: Record<string, unknown>) => {
      const r = (await c.callTool({ name, arguments: args })) as { content: Array<{ type: string; text?: string }> };
      return { json: JSON.parse(r.content.find((x) => x.type === "text")?.text ?? "{}") as Record<string, unknown>, images: r.content.filter((x) => x.type === "image").length };
    };
    const begin = await call("lr_begin_session", { intent_id: "landscape_golden_hour", mode: "variants" });
    const id = begin.json["session_id"] as string;
    for (const target of ["A", "B", "C"]) await call("lr_step", { session_id: id, target, settings: { clarity: 3 }, rationale: "dry run" });
    const pick = await call("lr_select_variant", { session_id: id, variant: "C" });
    const step = await call("lr_step", { session_id: id, settings: { vibrance: 3 }, rationale: "dry run" });
    const end = await call("lr_end_session", { session_id: id, outcome: "accept" });
    await c.close();
    return `chat: begin ok ${String(begin.json["ok"])} with ${begin.images} image(s), ${(begin.json["variants"] as unknown[] | undefined)?.length ?? 0} copies; pick ${String(pick.json["ok"])}; step ${String(step.json["pass"])}; end ${String(end.json["outcome"])}`;
  };
  const clickPreset = (name: string): void => {
    const target = lr.copies.get(lr.selected)?.settings ?? lr.settings;
    for (const [attr, raw] of presetDescription(parseXml(readFileSync(path.join(presets, `${name}.xmp`), "utf8"))).attrs) {
      const key = attr.replace(/^crs:/, "");
      if (attr.startsWith("crs:") && key in target) target[key] = typeof target[key] === "number" ? Number(raw) : raw;
    }
  };

  const cli = spawn(process.execPath, [path.join(repo, "engine/dist/devtools/phase4-check-cli.js")], { env, stdio: ["pipe", "pipe", "pipe"] });
  let out = "";
  const notes: string[] = [];
  cli.stderr.on("data", (d) => (out += String(d)));
  let pending = Promise.resolve();
  cli.stdout.on("data", (d) => {
    const text = String(d);
    out += text;
    pending = pending.then(async () => {
      const enter = (s = ""): void => void cli.stdin.write(`${s}\n`);
      if (/click 20260907-_OZ80099\.NEF \(the original/.test(text)) { lr.selected = lr.uuid; enter(); }
      else if (/Lightroom now shows copy/.test(text)) enter();
      else if (/Type y or n, then Enter:/.test(text)) enter("y");
      else if (/Which copy should the session continue on/.test(text)) enter("B");
      else if (/3\. Then press Enter here/.test(text)) { plugin.dropEventClient(); enter(); }
      else if (/click "(AVG P4check [^"]+)" once/.test(text)) { clickPreset(/click "(AVG P4check [^"]+)" once/.exec(text)?.[1] as string); enter(); }
      else if (/7\. Come back to this window/.test(text)) { notes.push(await chat()); enter(); }
      else if (/First quit Claude Desktop/.test(text)) enter();
      else if (/5\. Press Enter here|Still in the catalog/.test(text)) { lr.copies.clear(); enter(); }
      else if (/ {2}3\. Press Enter here/.test(text)) { for (const f of readdirSync(presets)) unlinkSync(path.join(presets, f)); enter(); }
    });
  });
  const code = await new Promise((r) => cli.on("close", r));
  await plugin.close();
  const transcript = out.split(os.homedir()).join("%USERPROFILE%");
  writeFileSync(path.join(repo, "logs", "scratch-p4-dryrun-output.txt"), `${transcript}\n[driver] ${notes.join("; ")}\n[driver] exit code ${String(code)}; History steps ${lr.history.length}; exports ${lr.exports}; copies left ${lr.copies.size}; preset files left ${readdirSync(presets).length}\n`);
  console.log(transcript);
  console.log(`[driver] ${notes.join("; ")}; exit ${String(code)}`);
  expect(code).toBe(0);
});
