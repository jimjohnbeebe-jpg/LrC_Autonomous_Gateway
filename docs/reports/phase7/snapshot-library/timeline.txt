--- mods-at.mjs ---
// node mods-at.mjs <deck log dir> <ISO time>...: for each time, the last Lightroom title the Deck logs saw at or
// before it (module, when, which log) and the first one after it, from the `windows` lines.
import fs from "node:fs";
import path from "node:path";
const [dir, ...times] = process.argv.slice(2);
const seen = [];
for (const f of fs.readdirSync(dir).sort()) for (const line of fs.readFileSync(path.join(dir, f), "utf8").split("\n")) {
  const m = line.match(/Lightroom Classic - (\w+)/);
  if (!m) continue;
  let e; try { e = JSON.parse(line); } catch { continue; }
  seen.push({ t: e.t, f, mod: m[1] });
}
seen.sort((a, b) => a.t - b.t);
const iso = (t) => new Date(t).toISOString();
for (const at of times) {
  const t = Date.parse(at);
  const before = seen.filter((s) => s.t <= t).at(-1);
  const after = seen.find((s) => s.t > t);
  console.log(`${at}  before: ${before ? `${before.mod} ${iso(before.t)} ${before.f}` : "-"}  after: ${after ? `${after.mod} ${iso(after.t)} ${after.f}` : "-"}`);
}
