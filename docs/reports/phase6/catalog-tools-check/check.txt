// engine's own Tools class (engine\dist), as Claude Desktop's engine would call them.
//   1. takes the engine instance lock; reads the selection (lr_get_selected_photos): the photo P;
//      reads P's Develop settings (the start);
//   2. searches: P by file name; with its rating; with capture dates around all time, before 1900

//   6. ALWAYS (finally): P's rating put back, the probe keyword taken off, and P's Develop settings
//      compared with the start (they must not change: ratings and keywords are not Develop settings).
// Collection names and P's keyword names are not recorded (Jim's own data): counts only.
// Results: <out>\check.json and the console. Usage: node check.mts <repo root>

const save = () => writeFileSync(path.join(outDir, "check.json"), JSON.stringify(result, null, 2));

const step = (name: string, data: Record<string, unknown>) => {

  result.checks[name] = ok;
  save();

};
/** A tool call, timed; its JSON, or its error body. */
const call = async (name: string, fn: () => Promise<{ json: Record<string, any> }>) => {
  const s = performance.now();
  try {
    const out = await fn();
    return { ok: true as const, ms: Math.round(performance.now() - s), json: out.json };
  } catch (err) {
    return { ok: false as const, ms: Math.round(performance.now() - s), error: toToolError(err).body() };
  }
};

const dev = devOverrides();
const lock = await acquireInstanceLock(dev.lockPort);
if (!lock.ok) {
  result.lock = { ok: false, port: lock.port, pid: lock.pid };
  save();
  say(`LOCK BUSY: another engine (pid ${lock.pid}) holds port ${lock.port}. Quit Claude Desktop, then run again.`);
  process.exit(2);
}
const client = new BridgeClient({ engineVersion: `${ENGINE_VERSION}-check`, log: (m: string) => say(`  (${m})`), ...dev.bridge });
client.start();
const hello = await client.waitConnected(30_000);
result.hello = { plugin_version: hello.plugin_version, lrc_version: hello.lrc_version };
if (hello.plugin_version !== PLUGIN) {
  say(`FAILED: Lightroom runs plugin ${hello.plugin_version}, not ${PLUGIN}. Restart Lightroom.`);
  result.failed = "plugin version";
  save();
  process.exit(1);
}
const map = loadDefaultParamMap();
const tools = new Tools({
  client,
  map,
  previews: new PreviewService(client),
  intents: new IntentLibrary({ map }),
  sessionLogDir: path.join(outDir, "sessions"),
  engineVersion: `${ENGINE_VERSION}-check`,
  ensureBridge: async () => void (await client.waitConnected(15_000)),
  log: new ToolLog(outDir),
  hud: false,
});

// 1. The selected photo.
const sel = await call("lr_get_selected_photos", () => tools.getSelectedPhotos({ limit: 5 }));
step("selection", { ms: sel.ms, ok: sel.ok, count: sel.ok ? sel.json.count : null, first: sel.ok ? sel.json.photos[0] : sel.error });
if (!sel.ok || !sel.json.photos[0]?.uuid) {
  say("FAILED: select one photo in Lightroom, then run again.");
  process.exit(1);
}
const P = sel.json.photos[0] as { uuid: string; filename: string; rating: number; capture_time: string | null };
const startRating = P.rating;
const settingsOf = async () => map.fromSdk((await client.request("get_settings", { photo_uuid: P.uuid })).settings).settings as Record<string, unknown>;
const start = await settingsOf();
say(`Photo: ${P.filename}, rating ${startRating}, captured ${P.capture_time}`);
check("the selection lists the photo with uuid, filename, rating and capture time", typeof P.filename === "string" && typeof P.capture_time === "string");

const found = async (label: string, args: Record<string, unknown>) => {
  const r = await call("lr_search_photos", () => tools.searchPhotos({ filename: P.filename, limit: 500, ...args }));
  const has = r.ok && (r.json.photos as Array<{ uuid: string }>).some((p) => p.uuid === P.uuid);
  step(`search ${label}`, { args, ms: r.ok ? r.ms : null, ok: r.ok, count: r.ok ? r.json.count : null, found: has, ...(r.ok ? {} : { error: r.error }) });
  return has;
};
const keywordOn = async () => found("by the probe keyword", { keywords: [KEYWORD] });

let ratingChanged = false;
let keywordAdded = false;
try {
  // 2. Searches.
  check("search by file name finds it", await found("by file name", {}));
  check("search by file name and its rating finds it", await found("with its rating", { rating: startRating }));
  check("search by file name and another rating does not", !(await found("with another rating", { rating: startRating === 5 ? 4 : 5 })));
  check("captured 1900-01-01..2100-12-31 finds it", await found("all time", { start_date: "1900-01-01", end_date: "2100-12-31" }));
  check("captured on or before 1900-01-01 does not", !(await found("before 1900", { end_date: "1900-01-01" })));
  check("captured on or after 2100-01-01 does not", !(await found("after 2100", { start_date: "2100-01-01" })));
  const all = await call("lr_search_photos", () => tools.searchPhotos({ limit: 1 }));
  step("search with no filter", { ms: all.ms, ok: all.ok, count: all.ok ? all.json.count : null, warning: all.ok ? all.json.warning : null, ...(all.ok ? {} : { error: all.error }) });
  check("a search with no filter answers, with the warning", all.ok && typeof all.json.warning === "string", all.ok ? `${all.json.count} photos, ${all.ms} ms` : JSON.stringify(all.error));

  // 3. Collections.
  const cols = await call("lr_list_collections", () => tools.listCollections({ limit: 500 }));
  step("collections", { ms: cols.ms, ok: cols.ok, count: cols.ok ? cols.json.count : null, smart: cols.ok ? cols.json.collections.filter((c: any) => c.smart).length : null, in_sets: cols.ok ? cols.json.collections.filter((c: any) => c.set !== null).length : null, ...(cols.ok ? {} : { error: cols.error }) });
  check("lr_list_collections answers", cols.ok, cols.ok ? `${cols.json.count} collections, ${cols.ms} ms` : JSON.stringify(cols.error));
  for (const smart of [false, true]) {
    const c = cols.ok ? cols.json.collections.find((x: any) => x.smart === smart && x.photo_count >= 1 && x.photo_count <= 500) : undefined;
    const label = smart ? "a smart collection" : "a plain collection";
    if (!c) {
      step(`collection search, ${label}`, { skipped: "none with 1-500 photos" });
      say(`--   ${label}: none with 1-500 photos, not checked`);
      continue;
    }
    const r = await call("lr_search_photos", () => tools.searchPhotos({ collection_id: c.id, limit: 500 }));
    step(`collection search, ${label}`, { photo_count: c.photo_count, in_set: c.set !== null, ms: r.ms, ok: r.ok, count: r.ok ? r.json.count : null, ...(r.ok ? {} : { error: r.error }) });
    check(`searching ${label} by id finds its photo_count`, r.ok && r.json.count === c.photo_count, r.ok ? `${r.json.count} of ${c.photo_count}` : JSON.stringify(r.error));
  }

  // 4. Rating.
  for (const rating of [startRating === 4 ? 2 : 4, 0, startRating]) {
    ratingChanged = true;
    const r = await call("lr_set_rating", () => tools.setRating({ uuids: [P.uuid], rating }));
    const photo = r.ok ? r.json.photos[0] : null;
    step(`rating ${rating}`, { ms: r.ms, ok: r.ok, photo, failed: r.ok ? r.json.failed : null, ...(r.ok ? {} : { error: r.error }) });
    check(`rating ${rating} written and read back`, r.ok && photo?.after === rating, r.ok ? `before ${photo?.before}, after ${photo?.after}, ${r.ms} ms` : JSON.stringify(r.error));
    check(`then a search by file name and rating ${rating} finds it`, await found(`with rating ${rating}`, { rating }));
  }
  ratingChanged = false;

  // 5. Keywords.
  if (await keywordOn()) {
    check(`the photo had no keyword "${KEYWORD}" before`, false);
  } else {
    keywordAdded = true;
    const add = await call("lr_set_keywords", () => tools.setKeywords({ uuids: [P.uuid], add: [KEYWORD] }));
    const a = add.ok ? add.json.photos[0] : null;
    step("keyword added", { ms: add.ms, ok: add.ok, before_count: a?.before.length ?? null, after_count: a?.after.length ?? null, failed: add.ok ? add.json.failed : null, ...(add.ok ? {} : { error: add.error }) });
    check("the keyword was added and read back", add.ok && a?.after.includes(KEYWORD) === true, add.ok ? `${a?.before.length} -> ${a?.after.length} keywords, ${add.ms} ms` : JSON.stringify(add.error));
    check("then a search by it finds the photo", await keywordOn());
    const rm = await call("lr_set_keywords", () => tools.setKeywords({ uuids: [P.uuid], remove: [KEYWORD] }));
    const b = rm.ok ? rm.json.photos[0] : null;
    step("keyword removed", { ms: rm.ms, ok: rm.ok, after_count: b?.after.length ?? null, failed: rm.ok ? rm.json.failed : null, ...(rm.ok ? {} : { error: rm.error }) });
    check("the keyword was removed, the photo's other keywords as before", rm.ok && JSON.stringify(b?.after) === JSON.stringify(a?.before), rm.ok ? `${b?.after.length} keywords, ${rm.ms} ms` : JSON.stringify(rm.error));
    keywordAdded = !rm.ok;
    check("then a search by it does not find the photo", !(await keywordOn()));
  }
} catch (err) {
  result.error = String(err);
  say(`ERROR: ${String(err)}`);
} finally {
  // 6. Put back.
  const back: Record<string, unknown> = {};
  if (ratingChanged) back["rating"] = await call("lr_set_rating", () => tools.setRating({ uuids: [P.uuid], rating: startRating }));
  if (keywordAdded) back["keyword"] = await call("lr_set_keywords", () => tools.setKeywords({ uuids: [P.uuid], remove: [KEYWORD] }));
  const now = await call("lr_get_selected_photos", () => tools.getSelectedPhotos({ limit: 5 }));
  const end = await settingsOf();
  const differing = [...new Set([...Object.keys(start), ...Object.keys(end)])].filter((k) => JSON.stringify(start[k]) !== JSON.stringify(end[k]));
  const nowP = now.ok ? (now.json.photos as Array<{ uuid: string; rating: number }>).find((p) => p.uuid === P.uuid) : undefined;
  step("put back", { extra_calls: back, rating_now: nowP?.rating ?? null, develop_settings_differing: differing });
  check("the photo's rating is its own again", nowP?.rating === startRating, `${nowP?.rating} (was ${startRating})`);
  check("no Develop setting changed", differing.length === 0, differing.join(", "));
  check("the probe keyword is off the photo", !(await keywordOn()));
  const failedChecks = Object.entries(result.checks).filter(([, ok]) => !ok).map(([name]) => name);
  result.ended = at();
  result.failed_checks = failedChecks;
  save();
  say(failedChecks.length === 0 ? "Catalog tools check: WORKED" : `Catalog tools check: FAILED (${failedChecks.length}): ${failedChecks.join("; ")}`);
  say(`Results: ${path.join(outDir, "check.json")}`);
  client.stop();
  await lock.lock.release();
  process.exit(failedChecks.length === 0 ? 0 : 1);
}
