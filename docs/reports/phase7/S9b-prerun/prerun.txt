--- Appendix I: zz-drag.ts ---
import koffi from "koffi";
import { execSync, spawn } from "node:child_process";
import { hudLog, hudPids, killTree, now, pollUntil, sleep, startStub } from "./harness.ts";
import { deckRect, mainWindow, pidsByExe, rect, tree, visible } from "./win32.ts";
const u = koffi.load("user32.dll");
const SetWindowPos = u.func("bool __stdcall SetWindowPos(uintptr_t h, uintptr_t after, int x, int y, int cx, int cy, uint32_t f)");
process.env.LRC_AVG_S9_LR_EXE = "notepad.exe";
spawn("notepad.exe", [], { detached: true, stdio: "ignore" }).unref();
await sleep(2500);
const np = pidsByExe("notepad.exe")[0] ?? 0; const npH = mainWindow(np);
const stub = await startStub();
stub.cmd({ cmd: "spawn" });
const sp = await stub.waitFor((e) => e.ev === "spawned", 5000);
const pid = Number(sp?.pid);
await stub.waitFor((e) => e.ev === "paint", 15000);
await sleep(800);
const hud = Number(hudLog(pid, sp?.t ?? 0).find((e) => e.ev === "start")?.hwnd);
const eq = (a: any, b: any) => JSON.stringify(a) === JSON.stringify(b);
const r0 = rect(npH);
// a single move: should stay visible, in place
let t = now(); SetWindowPos(npH, 0, r0.left + 100, r0.top, r0.right - r0.left, r0.bottom - r0.top, 0x14);
let ok = await pollUntil(() => visible(hud) && eq(rect(hud), deckRect(npH)), 1000, 2);
console.log("single move: in place, visible ms", ok ? (ok - t).toFixed(1) : null);
await sleep(600);
// a burst of 10 moves, 60 ms apart (like Jim's drag cadence)
let hiddenSeen = false, detachedPolls = 0;
const t0 = now();
for (let i = 0; i < 10; i++) {
  SetWindowPos(npH, 0, r0.left + 100 + i * 20, r0.top + i * 5, r0.right - r0.left, r0.bottom - r0.top, 0x14);
  const until = now() + 60;
  while (now() < until) { if (!visible(hud)) hiddenSeen = true; else if (!eq(rect(hud), deckRect(npH))) detachedPolls++; await sleep(2); }
}
const tEnd = now();
ok = await pollUntil(() => visible(hud) && eq(rect(hud), deckRect(npH)), 2000, 2);
console.log("burst: hidden during burst", hiddenSeen, "detached polls", detachedPolls, "back in place ms after last move", ok ? (ok - tEnd).toFixed(1) : null);
console.log(hudLog(pid, sp?.t ?? 0).filter((e) => e.ev.startsWith("move_") || e.ev === "burst_quiet").map((e) => JSON.stringify(e)).join("\n"));
stub.cmd({ cmd: "quit" });
for (const p of hudPids()) await killTree(tree(p));
execSync(`taskkill /PID ${np} /F`);
process.exit(0);

--- Appendix J: zz-follow8.ts (run as: ( sleep 9; echo; sleep 12; echo; sleep 3; for i in 1 2 3 4 5 6; do echo y; sleep 0.3; done ) | node zz-follow8.ts; run 4a used setTimeout 3000 and sleep 9 between the Enters) ---
import koffi from "koffi";
import { execSync, spawn } from "node:child_process";
import { closePrompts, hudPids, killTree, saver, sleep, startStub, hudLog, type Results } from "./harness.ts";
import { followChecks } from "./jim.ts";
import type { Ctx } from "./auto.ts";
import { mainWindow, pidsByExe, rect, tree } from "./win32.ts";
const u = koffi.load("user32.dll");
const SetWindowPos = u.func("bool __stdcall SetWindowPos(uintptr_t h, uintptr_t after, int x, int y, int cx, int cy, uint32_t f)");
const ShowWindow = u.func("bool __stdcall ShowWindow(uintptr_t h, int cmd)");
process.env.LRC_AVG_S9_LR_EXE = "notepad.exe";
spawn("notepad.exe", [], { detached: true, stdio: "ignore" }).unref();
await sleep(2500);
const np = pidsByExe("notepad.exe")[0] ?? 0; const npH = mainWindow(np);
const stub = await startStub();
stub.cmd({ cmd: "spawn" });
const sp = await stub.waitFor((e) => e.ev === "spawned", 5000);
const pid = Number(sp?.pid);
await stub.waitFor((e) => e.ev === "paint", 15000);
await sleep(800);
const hwnd = Number(hudLog(pid, sp?.t ?? 0).find((e) => e.ev === "start")?.hwnd);
const r: Results = { run_id: "zz-follow8", errors: [] };
const ctx: Ctx = { stub, r, save: saver(r), lr: { pid: np, hwnd: npH }, hud: { pid, hwnd } };
const r0 = rect(npH);
// the "user": after the first Enter (stdin at ~2 s), a drag-like burst, a single jump, a minimise/restore
setTimeout(async () => {
  for (let i = 0; i < 12; i++) { SetWindowPos(npH, 0, r0.left + i * 15, r0.top, r0.right - r0.left, r0.bottom - r0.top, 0x14); await sleep(64); }
  await sleep(1200);
  SetWindowPos(npH, 0, r0.left, r0.top, r0.right - r0.left - 200, r0.bottom - r0.top, 0x14);
  await sleep(1200);
  ShowWindow(npH, 6); await sleep(1000); ShowWindow(npH, 9);
}, 11000);
await followChecks(ctx);
console.log(JSON.stringify((r.s9_8 as any).episodes.map((e: any) => ({ k: e.kind, settled: e.settled, ms: e.ms === null ? null : Math.round(e.ms), detached: e.detached_ms, changes: e.lr_changes, span: e.lr_change_span_ms }))));
console.log("move_events", JSON.stringify((r.s9_8 as any).move_events), "answers", (r.s9_8 as any).jim_drag_hidden, (r.s9_8 as any).jim_text_sharp);
closePrompts();
stub.cmd({ cmd: "quit" });
for (const p of hudPids()) await killTree(tree(p));
execSync(`taskkill /PID ${np} /F`);
process.exit(0);
