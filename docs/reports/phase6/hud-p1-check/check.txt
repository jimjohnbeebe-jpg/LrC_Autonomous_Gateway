      child.stdin.write(`${kind === "press Enter" ? "" : kind === "y/n" ? opts.answer(m[1] ?? "", asked++) : "b"}\n`);
    }
  });
  const code: number = await new Promise((resolve) => child.on("close", (c) => resolve(c ?? -1)));
  await plugin.close();
  return { out, code, hud, check: JSON.parse(readFileSync(path.join(dir, "check.json"), "utf8")), received: plugin.received.map((r) => r.name) };
}

describe("hud-p1 check dry run", () => {
  it("all yes: WORKED", async () => {
    const r = await run({ version: "0.9.0", answer: () => "y" });
    log(r.out);
    log("exit", r.code, "updates taken", r.hud.taken.map((u) => u.stage).join(","), "refused", r.hud.refused.length, "notTaken", r.hud.notTaken.length);
    log("writes:", r.received.filter((n) => !["hello", "ping", "get_selection", "get_context", "hud_update"].includes(n)));
    expect(r.out).toContain("WORKED");
    expect(r.code).toBe(0);
    expect(r.hud.last()?.stage).toBe("ended");
  }, 120_000);
  it("wrong plugin version: FAILED", async () => {
    const r = await run({ version: "0.8.0", answer: () => "y" });
    log(r.out);
    expect(r.out).toContain("FAILED");
    expect(r.code).toBe(1);
  }, 60_000);
  it("an n answer: FAILED, still ends", async () => {
    const r = await run({ version: "0.9.0", answer: (_q, n) => (n === 0 ? "n" : "y") });
    log(r.out.split("\n").slice(-12).join("\n"));
    expect(r.out).toContain("FAILED");
    expect(r.code).toBe(1);
    expect(r.hud.last()?.stage).toBe("ended");
  }, 120_000);
});
