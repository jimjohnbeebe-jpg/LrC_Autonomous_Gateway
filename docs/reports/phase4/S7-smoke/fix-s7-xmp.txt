-- io shim over FS
local function fileObj(path, mode)
    local buf, pos = (mode:sub(1, 1) == "a") and (FS[path] or "") or "", 1
    if mode:sub(1, 1) == "r" then buf = FS[path] end
    local f = {}
    function f:write(...)
        if FLAGS.write_fails and path:find("AVG S7 xmp", 1, true) then return nil, "No space left on device (fake)" end
        for _, s in ipairs({ ... }) do buf = buf .. tostring(s) end FS[path] = buf return self
    end
    function f:read(fmt) return buf end
    function f:lines()
        local lines = {}
        for l in (buf .. "\n"):gmatch("(.-)\n") do table.insert(lines, l) end
        if lines[#lines] == "" then table.remove(lines) end
        local i = 0
        return function() i = i + 1 return lines[i] end
    end
    function f:close() if mode:sub(1, 1) ~= "r" then FS[path] = buf end return true end
    return f
end
io.open = function(path, mode)
    mode = mode or "r"
    if mode:sub(1, 1) == "r" and FS[path] == nil then return nil, path .. ": No such file" end
    if mode:sub(1, 1) == "w" then FS[path] = "" end
    return fileObj(path, mode)
end
