== Appendix: s9b-analysis.ts, verbatim

// S9b S9-8 analysis: the harness's episodes beside the HUD's own log. Run from the repo root:
//   node <this file>
import { readFileSync } from "node:fs";

const dir = "docs/reports/phase7/S9b/";
const run = JSON.parse(readFileSync(dir + "s9_s9b-2026-10-05T11-16-10-168Z.json", "utf8"));
const log = readFileSync(dir + "hud_24220_1791198983874.jsonl", "utf8").trim().split("\n").map((l) => JSON.parse(l));
const S = Date.parse(run.started);
const s = (t: number): string => ((t - S) / 1000).toFixed(3);
const box = (r: { left: number; top: number; right: number; bottom: number } | null): string => (r ? `${r.left},${r.top},${r.right},${r.bottom}` : "-");

console.log(`== A. The harness's S9-8 episodes (s9_8.episodes), in order; run started ${run.started}`);
run.s9_8.episodes.forEach((e: Record<string, any>, i: number) =>
  console.log(`${String(i + 1).padStart(2)} ${e.kind.padEnd(9)} ms=${e.ms === null ? "null" : e.ms.toFixed(1)} detached_ms=${e.detached_ms} lr_changes=${e.lr_changes} span=${e.lr_change_span_ms} hud_visible=${e.hud_visible} lr=${box(e.lr_rect)} hud=${box(e.hud_rect)} expected=${box(e.expected)}`));
console.log(`move_events ${JSON.stringify(run.s9_8.move_events)}`);

console.log("\n== B. Each movesize hide/show pair in the HUD log (seconds from the run start)");
let hide: number | null = null;
let follows: number[] = [];
let places: { t: number; got: number[] }[] = [];
for (const e of log) {
  if (e.ev === "move_hide") [hide, follows, places] = [e.t, [], []];
  else if (e.ev === "follow" && hide !== null) follows.push(e.t_event);
  else if (e.ev === "place" && hide !== null) places.push(e);
  else if (e.ev === "move_show" && hide !== null) {
    const gaps = follows.slice(1).map((x, i) => x - follows[i]!);
    const last = places.at(-1)!;
    console.log(`hide ${s(hide)} show ${s(e.t)} hidden ${(e.t - hide).toFixed(0)} ms; location changes ${follows.length}, largest gap ${gaps.length ? Math.max(...gaps).toFixed(0) : "-"} ms; last change -> show ${follows.length ? (e.t - follows.at(-1)!).toFixed(0) : "-"} ms; final place -> show ${(e.t - last.t).toFixed(1)} ms, got [${last.got}]`);
    hide = null;
  }
}

console.log("\n== C. HUD log lines from 360 s to 395 s (minimise, restore, screen modes); webview lines left out");
for (const e of log) {
  const r = (e.t - S) / 1000;
  if (r < 360 || r > 395 || e.ev === "webview") continue;
  const { ev, t, ...rest } = e;
  console.log(`${s(t)} ${ev} ${JSON.stringify(rest)}`);
}
