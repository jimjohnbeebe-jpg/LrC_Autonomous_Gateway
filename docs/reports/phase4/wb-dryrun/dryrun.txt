Appendix: the driver, verbatim (wb-dryrun.ts, run with Node's type stripping)
-----------------------------------------------------------------------------
// SCRATCH: dry run of the built `engine\dist\devtools\wb-check-cli.js` against a scratch plugin on
// ports 18765/18766 (lock 18767) with its own token file, so Jim's running Lightroom is not touched.
// TEMP/TMP point at the scratch folder, so the results land there. Usage (repo root):
//   node <this file> <scratch dir> [refuse]
// "refuse": the scratch plugin drops WhiteBalance from writes, as Lightroom refusing "Custom" would.
import { spawn } from "node:child_process";
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import net from "node:net";
import path from "node:path";

const repo = process.cwd();
const scratch = process.argv[2] as string;
const refuse = process.argv[3] === "refuse";
mkdirSync(scratch, { recursive: true });
const tokenFile = path.join(scratch, "dry_token");
const TOKEN = "dry-token-wb";
writeFileSync(tokenFile, TOKEN);

const dump = (JSON.parse(readFileSync(path.join(repo, "docs/reports/phase0/S5/s5_20260907-_OZ80093.NEF.json"), "utf8")) as { settings: Record<string, unknown> }).settings;
let settings: Record<string, unknown> = structuredClone(dump);
const snaps = new Map<string, Record<string, unknown>>();
const log: string[] = [];
const UUID = "DRY-UUID";
const luaize = (v: unknown): unknown =>
  Array.isArray(v) ? v.map(luaize) : v && typeof v === "object" ? (Object.keys(v).length === 0 ? [] : Object.fromEntries(Object.entries(v).map(([k, x]) => [k, luaize(x)]))) : v;
let eventClient: net.Socket | null = null;
const send = (o: unknown): void => void eventClient?.write(JSON.stringify(o) + "\n");
const handlers: Record<string, (p: Record<string, unknown>) => unknown> = {
  hello: () => ({ protocol: 1, plugin_version: "0.4.0", lrc_version: "15.5.1", sdk_declared: 13, ports: { receive: 18765, send: 18766 } }),
  ping: (p) => ({ pong: true, nonce: p["nonce"] }),
  get_context: () => ({ uuid: UUID, local_id: 1, lrc_version: "15.5.1", filename: "20260907-_OZ80099.NEF", is_virtual_copy: false, file_format: "RAW" }),
  get_settings: () => ({ uuid: UUID, settings: luaize(settings) }),
  apply_settings: (p) => {
    for (const [k, v] of Object.entries(p["settings"] as Record<string, unknown>)) if (!(refuse && k === "WhiteBalance")) settings[k] = structuredClone(v);
    return { uuid: UUID, apply_ms: 30, read_ms: 300, command_ms: 340, read_back: luaize(settings) };
  },
  create_snapshot: (p) => {
    const id = `S${snaps.size + 1}`;
    snaps.set(id, structuredClone(settings));
    return { uuid: UUID, snapshot_id: id, id_global: "G", name: p["name"], same_name_count: 1 };
  },
  apply_snapshot: (p) => {
    settings = structuredClone(snaps.get(String(p["snapshot_id"])) ?? {});
    return { uuid: UUID, read_back: luaize(settings) };
  },
};

const commandServer = net.createServer((socket) => {
  socket.setEncoding("utf8");
  let buffer = "";
  socket.on("data", (chunk: string) => {
    buffer += chunk;
    let i: number;
    while ((i = buffer.indexOf("\n")) !== -1) {
      const msg = JSON.parse(buffer.slice(0, i)) as { id: string; name: string; token?: string; payload?: Record<string, unknown> };
      buffer = buffer.slice(i + 1);
      log.push(`${msg.name} ${JSON.stringify(msg.payload ?? {})}`);
      const handler = handlers[msg.name];
      if (msg.token !== TOKEN) send({ id: msg.id, type: "res", name: msg.name, ok: false, error: { code: "unauthorized", message: "token", recoverable: true } });
      else if (!handler) send({ id: msg.id, type: "res", name: msg.name, ok: false, error: { code: "unknown_command", message: msg.name, recoverable: false } });
      else send({ id: msg.id, type: "res", name: msg.name, ok: true, payload: handler(msg.payload ?? {}) });
    }
  });
  socket.on("error", () => {});
});
const eventServer = net.createServer((socket) => {
  eventClient = socket;
  socket.on("error", () => {});
});
await new Promise<void>((r) => commandServer.listen(18765, "127.0.0.1", () => r()));
await new Promise<void>((r) => eventServer.listen(18766, "127.0.0.1", () => r()));

const env = { ...process.env, LRC_AVG_COMMAND_PORT: "18765", LRC_AVG_EVENT_PORT: "18766", LRC_AVG_TOKEN_FILE: tokenFile, TEMP: scratch, TMP: scratch };
const child = spawn(process.execPath, [path.join(repo, "engine/dist/devtools/wb-check-cli.js")], { env, stdio: ["pipe", "pipe", "pipe"] });
let out = "";
child.stdout.on("data", (d: Buffer) => void (out += d.toString()));
child.stderr.on("data", (d: Buffer) => void (out += d.toString()));
// Enter after the click; y (As Shot), y (Temp slider), then the Custom question; with "refuse", n and the drag's Enter.
child.stdin.write(refuse ? "\ny\ny\nn\n\n" : "\ny\ny\ny\n");
child.stdin.end();
const code = await new Promise<number | null>((r) => child.on("close", r));
const same = JSON.stringify(settings) === JSON.stringify(dump);
console.log(out);
console.log(`exit code: ${code}`);
console.log(`scratch plugin commands:\n  ${log.join("\n  ")}`);
console.log(`scratch photo back to its start settings: ${same}`);
commandServer.close();
eventServer.close();
eventClient?.destroy();
