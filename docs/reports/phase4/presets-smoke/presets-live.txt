== Appendix A: profiles.mjs ==
// Read-only: each candidate photo's camera profile and edit count, by uuid (plugin 0.4.0 photo_uuid).
import { BridgeClient } from "<repo>/engine/dist/bridge/index.js";
import { acquireInstanceLock, BridgeGate, devOverrides, ENGINE_VERSION } from "<repo>/engine/dist/mcp/index.js";
import { loadDefaultParamMap } from "<repo>/engine/dist/params/index.js";
const uuids = { /* the five uuid -> file name pairs of step 2, from the Phase 3 session logs */ };
const dev = devOverrides();
const client = new BridgeClient({ engineVersion: ENGINE_VERSION, log: () => {}, ...dev.bridge });
const gate = new BridgeGate(client, () => acquireInstanceLock(dev.lockPort));
const map = loadDefaultParamMap();
try {
  await gate.ready();
  for (const [uuid, name] of Object.entries(uuids)) {
    const r = await client.request("get_settings", { photo_uuid: uuid });
    const read = map.fromSdk(r.settings);
    const nz = Object.entries(read.settings).filter(([, v]) => typeof v === "number" && v !== 0).map(([k]) => k);
    const curves = Object.entries(read.settings).filter(([k, v]) => k.startsWith("tone_curve.") && Array.isArray(v) && v.length > 4).map(([k]) => k);
    console.log(name, "|", read.camera_profile.name, "| look:", read.camera_profile.look_name, "| nonzero:", nz.length, "| curves:", curves.join(","), "| WB:", r.settings.WhiteBalance);
  }
} finally {
  await gate.release();
}
