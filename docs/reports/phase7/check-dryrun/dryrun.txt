  });
  const click = (name: "hud_abort" | "hud_accept" | "hud_pick" | "hud_approve_pass", extra: Record<string, unknown> = {}): void => {
    const last = lr.hud.last();
    if (last) hudEvent(plugin, name, { session_id: last.session_id, seq_seen: last.seq, source: "hud", ...extra });
  };
  const fakeLr: ChildProcess = spawn("ping", ["-n", "100000", "127.0.0.1"], { windowsHide: true, stdio: "ignore" });
  const env: Record<string, string> = {
    ...(process.env as Record<string, string>),
    LRC_AVG_COMMAND_PORT: String(plugin.commandPort),
    LRC_AVG_EVENT_PORT: String(plugin.eventPort),
    LRC_AVG_TOKEN_FILE: tokenFile,
    LRC_AVG_P7_OUT: path.join(scratch, "out"),
    LRC_AVG_PLUGIN_LOG: bridgeLog,
    LRC_AVG_HUD_EXE: dummyExe,
    LRC_AVG_HUD_LR_EXE: "PING.EXE",
    LOCALAPPDATA: path.join(scratch, "localappdata"),
  };
  delete env["LRC_AVG_LOG_DIR"];

  // "Claude Desktop": the real engine over MCP.
  let desk: Client | null = null;
  const notes: string[] = [];
  const call = async (name: string, args: Record<string, unknown>): Promise<Record<string, unknown>> => {
    if (!desk) {
      desk = new Client({ name: "dry-desktop", version: "0" });
      await desk.connect(new StdioClientTransport({ command: process.execPath, args: [path.join(repo, "engine/dist/mcp/main.js")], env, stderr: "pipe" }));
    }
    const r = (await desk.callTool({ name, arguments: args }, undefined, { timeout: 180000 })) as { content: Array<{ type: string; text?: string }> };
    return JSON.parse(r.content.find((x) => x.type === "text")?.text ?? "{}") as Record<string, unknown>;
  };
  const quitDesktop = async (): Promise<void> => {
    await (desk as Client | null)?.close();
    desk = null;
  };
  const chat1 = async (): Promise<void> => {
    const begin = await call("lr_begin_session", { intent_id: "landscape_golden_hour", return_image: "none" });
    const sid = String(begin["session_id"]);
    await call("lr_step", { session_id: sid, settings: { vibrance: 3 }, rationale: "dry run", return_image: "none" });
    const end = await call("lr_end_session", { session_id: sid, outcome: "accept" });
    notes.push(`chat 1: begin ${String(begin["ok"] ?? sid)}, end ${String(end["outcome"])}`);
  };
  const chat2 = async (): Promise<void> => {
    const begin = await call("lr_begin_session", { intent_id: "landscape_golden_hour", return_image: "none" });
    await call("lr_step", { session_id: String(begin["session_id"]), settings: { vibrance: 3 }, rationale: "dry run", return_image: "none" });
    await quitDesktop();
    notes.push("chat 2: one pass, then Claude Desktop quit mid-edit");
  };

  const runCheck = async (args: string[]): Promise<{ code: unknown; out: string }> => {
    const cli = spawn(process.execPath, [path.join(repo, "hud/probe/check/check.ts"), ...args], { env, stdio: ["pipe", "pipe", "pipe"] });
    let out = "";
    let tail = "";
    cli.stderr.on("data", (d) => (out += String(d)));
    const enter = (s = ""): void => void cli.stdin.write(`${s}\n`);
    const later = (ms: number, fn: () => void): void => void setTimeout(fn, ms);
    const says = (line: string): void => {
      if (/Then click Accept in the Deck/.test(line)) later(1000, () => click("hud_accept"));
      if (/then press Enter on your keyboard/.test(line)) later(1000, () => click("hud_approve_pass", { pass: 1 }));
      if (/press Ctrl\+Backspace twice/.test(line)) later(1000, () => click("hud_abort"));
      if (/press 3, then Enter/.test(line)) later(1000, () => click("hud_pick", { variant: "C" }));
      if (/Click its Abort button/.test(line)) later(1000, () => click("hud_abort"));
      if (/Now quit Lightroom/.test(line)) later(2000, () => void fakeLr.kill());
    };
    let pending = Promise.resolve();
    let recent: string[] = []; // the instruction's lines before its prompt line
    const prompt = async (text: string): Promise<boolean> => {
      if (/Type y or n, then Enter:\s*$/.test(text)) enter("y");
      else if (/Press Enter when done\.\s*$/.test(text)) {
        const mode = /set Mode to "(Approve each pass|Autonomous)"/.exec(text);
        if (mode && lr.prefs) lr.prefs["mode"] = mode[1] === "Autonomous" ? "autonomous" : "approve_each_pass";
        if (/Remove the \d+ copies/.test(text)) lr.copies.clear();
        if (/Start Claude Desktop/.test(text)) await chat1();
        if (/As soon as the Deck appears, quit Claude Desktop/.test(text)) await chat2();
        if (/still holds the bridge/.test(text)) await quitDesktop();
        enter();
        if (/click Abort in the Deck at once/.test(text)) later(1500, () => click("hud_abort"));
      } else return false;
      return true;
    };
    cli.stdout.on("data", (d) => {
      const text = String(d);
      out += text;
      pending = pending.then(async () => {
        const lines = (tail + text).split("\n");
        tail = lines.pop() ?? "";
        for (const l of lines) {
          says(l);
          recent.push(l);
        }
        if (tail && (await prompt([...recent, tail].join("\n")))) {
          tail = "";
          recent = [];
        }
      });
    });
    const code = await new Promise((r) => cli.on("close", r));
    return { code, out };
  };

  const first = await runCheck([]);
  if (!fakeLr.killed) fakeLr.kill();
  const fakeLr2 = spawn("ping", ["-n", "100000", "127.0.0.1"], { windowsHide: true, stdio: "ignore" });
  const second = await runCheck(["--redo", process.env["P7DRY_REDO"] ?? "E5"]);
  fakeLr2.kill();
  await quitDesktop();
  await plugin.close();
  const scrub = (s: string): string => s.split(os.homedir()).join("%USERPROFILE%");
  writeFileSync(
    path.join(repo, "logs", "scratch-p7-dryrun-output.txt"),
    `== run 1 (exit ${String(first.code)}) ==\n${scrub(first.out)}\n== run 2: --redo E5 (exit ${String(second.code)}) ==\n${scrub(second.out)}\n[driver] ${notes.join("\n[driver] ")}\n[driver] exe restored: ${String(existsSync(dummyExe))}; copies left ${lr.copies.size}; scratch ${scrub(scratch)}\n`,
  );
  expect(first.code).toBe(0);
  expect(existsSync(dummyExe)).toBe(true);
});
