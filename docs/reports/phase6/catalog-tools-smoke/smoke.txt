== Appendix: fake-lr.lua (verbatim)

-- A fake Lightroom SDK for the catalog commands (Library.lua, Photos.lua) under fengari. Not
-- Lightroom: it checks the rules the plugin must follow (rule 03) and records what was called.
--   * a catalog query (findPhotos, getAllPhotos, getCollectionByLocalIdentifier, findPhotoByUuid)
--     inside a read or write gate is a violation;
--   * a metadata read outside a gate, or a write (setRawMetadata, addKeyword, removeKeyword)
--     outside a write gate, is a violation; gates may not nest;
--   * setRawMetadata("rating", 0) raises, as Automaat says the SDK does (upstream claim).
-- Its findPhotos matches filename "any", rating "==", keywords "all", captureTime ">" "<" "in".

local F = { violations = {}, gates = {}, yields = 0, created = 0, calls = {} }
local gate = nil -- "read" or "write"

local function violation(what) F.violations[#F.violations + 1] = what end
local function query(name) F.calls[#F.calls + 1] = name; if gate then violation(name .. " inside a " .. gate .. " gate") end end
local function needGate(what) if not gate then violation(what .. " outside a gate") end end
local function needWrite(what) if gate ~= "write" then violation(what .. " outside a write gate") end end

local keywords = {}
local function keyword(name)
    if not keywords[name] then
        keywords[name] = { name = name, getName = function(self) needGate("getName"); return self.name end }
    end
    return keywords[name]
end

local Photo = {}
Photo.__index = Photo
function Photo:getRawMetadata(key)
    needGate("getRawMetadata " .. key)
    if key == "uuid" then return self.uuid end
    if key == "isVirtualCopy" then return false end
    if key == "masterPhoto" then return self end
    if key == "rating" then return self.rating end
    if key == "keywords" then
        local list = {}
        for i, k in ipairs(self.keywords) do list[i] = k end
        return list
    end
    error("fake: unknown raw key " .. key)
end
function Photo:getFormattedMetadata(key)
    needGate("getFormattedMetadata " .. key)
    if key == "fileName" then return self.fileName end
    if key == "copyName" then return nil end
    if key == "dateTimeOriginal" then return self.day .. " 12:00:00" end
    error("fake: unknown formatted key " .. key)
end
function Photo:setRawMetadata(key, value)
    needWrite("setRawMetadata")
    if key ~= "rating" then error("fake: only rating") end
    if value == 0 then error("fake: rating 0 rejected (use nil)") end
    self.rating = value
end
function Photo:addKeyword(k)
    needWrite("addKeyword")
    for _, have in ipairs(self.keywords) do if have == k then return end end
    self.keywords[#self.keywords + 1] = k
end
function Photo:removeKeyword(k)
    needWrite("removeKeyword")
    for i, have in ipairs(self.keywords) do if have == k then table.remove(self.keywords, i) return end end
end

local function photo(id, uuid, fileName, rating, kws, day)
    local p = setmetatable({ localIdentifier = id, uuid = uuid, fileName = fileName, rating = rating, keywords = {}, day = day }, Photo)
    for _, n in ipairs(kws) do p.keywords[#p.keywords + 1] = keyword(n) end
    return p
end

local function collection(id, name, smart, members)
    return { localIdentifier = id, smart = smart, members = members,
        type = function() needGate("type"); return smart and "LrSmartCollection" or "LrCollection" end,
        getName = function() needGate("getName"); return name end,
        getPhotos = function() needGate("getPhotos"); return members end,
        isSmartCollection = function() needGate("isSmartCollection"); return smart end }
end
local function set(id, name, collections, sets)
    return { localIdentifier = id,
        type = function() needGate("type"); return "LrCollectionSet" end,
        getName = function() needGate("getName"); return name end,
        getChildCollections = function() needGate("getChildCollections"); return collections end,
        getChildCollectionSets = function() needGate("getChildCollectionSets"); return sets end }
end
local P = F.photos
local birds = collection(501, "Birds", false, { P[2], P[3] })
local fives = collection(502, "Five stars", true, { P[3], P[4] })
local y2026 = set(602, "2026", { fives }, {})
local best = set(601, "Best", {}, { y2026 })
local byId = { [501] = birds, [502] = fives, [601] = best, [602] = y2026 }

local function matches(p, c)
    local v = c.value
    if c.criteria == "filename" and c.operation == "any" then return p.fileName:find(v, 1, true) ~= nil end
    if c.criteria == "rating" and c.operation == "==" then return (p.rating or 0) == v end
    if c.criteria == "keywords" and c.operation == "all" then
        for _, k in ipairs(p.keywords) do if k.name == v then return true end end
        return false
    end
    if c.criteria == "captureTime" and c.operation == ">" then return p.day > v end
    if c.criteria == "captureTime" and c.operation == "<" then return p.day < v end
    if c.criteria == "captureTime" and c.operation == "in" then return p.day >= v and p.day <= c.value2 end
    error("fake: unknown criterion " .. tostring(c.criteria) .. " " .. tostring(c.operation))
end

F.catalog = {
    findPhotos = function(self, args)
        query("findPhotos")
        local desc = args.searchDesc
        if desc.combine ~= "intersect" then violation("findPhotos without combine = intersect") end
        local out = {}
        for _, p in ipairs(P) do
            local all = true
            for _, c in ipairs(desc) do if not matches(p, c) then all = false end end
            if all then out[#out + 1] = p end
        end
        return out
    end,
    getAllPhotos = function() query("getAllPhotos"); return P end,
    getCollectionByLocalIdentifier = function(self, id) query("getCollectionByLocalIdentifier"); return byId[id] end,
    findPhotoByUuid = function(self, uuid)
        query("findPhotoByUuid")
        for _, p in ipairs(P) do if p.uuid == uuid then return p end end
        return nil
    end,
    getChildCollections = function() needGate("getChildCollections"); return { birds } end,
    getChildCollectionSets = function() needGate("getChildCollectionSets"); return { best } end,
    createKeyword = function(self, name, synonyms, includeOnExport, parent, returnExisting)
        needWrite("createKeyword")
        if keywords[name] and returnExisting ~= true then error("fake: keyword exists") end
        if not keywords[name] then F.created = F.created + 1 end
        return keyword(name)
    end,
    withReadAccessDo = function(self, fn)
        if gate then violation("read gate inside a " .. gate .. " gate") end
        gate = "read"; fn(); gate = nil
    end,
    withWriteAccessDo = function(self, name, fn)
        if gate then violation("write gate inside a " .. gate .. " gate") end
        F.gates[#F.gates + 1] = name
        gate = "write"; fn(); gate = nil
    end,
}

local modules = {
    LrApplication = { activeCatalog = function() return F.catalog end },
    LrTasks = { pcall = pcall, yield = function()
        if gate then violation("yield inside a " .. gate .. " gate") end
        F.yields = F.yields + 1
    end },
}
function import(name) return modules[name] or error("fake: no module " .. name) end

return F

== Appendix: mutants.mts (verbatim)

// Mutants of Library.lua: each copies the plugin's Library.lua and Photos.lua into mut/<id>/, makes
// one change, runs cases.lua against it, and prints the checks that failed. A mutant is caught when
// at least one check fails. Usage: node mutants.mts <plugin folder>
import { execFileSync } from "node:child_process";
import { copyFileSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const here = path.dirname(fileURLToPath(import.meta.url));
const plugin = process.argv[2] ?? "";
const MUTANTS: Array<[string, string, string]> = [
  ["M1 findPhotos inside a read gate", "matches = catalog:findPhotos { searchDesc = desc } or {}",
    "catalog:withReadAccessDo(function() matches = catalog:findPhotos { searchDesc = desc } or {} end)"],
  ["M2 rating 0 written as 0", 'photo:setRawMetadata("rating", rating > 0 and rating or nil)', 'photo:setRawMetadata("rating", rating)'],
  ["M3 no yield between photos", "if i > offset + 1 then LrTasks.yield() end", ""],
  ["M4 collection filter ignores the criteria", "matches = matches and intersect(matches, inCollection) or inCollection", "matches = inCollection"],
  ["M5 createKeyword without returnExisting", "catalog:createKeyword(name, {}, true, nil, true)", "catalog:createKeyword(name, {}, true, nil, false)"],
  ["M6 rating written even when unchanged", "if before ~= rating then", "if true then"],
];
let caught = 0;
for (const [id, from, to] of MUTANTS) {
  const dir = path.join(here, "mut", id.split(" ")[0] as string);
  mkdirSync(dir, { recursive: true });
  for (const f of ["Library.lua", "Photos.lua"]) copyFileSync(path.join(plugin, f), path.join(dir, f));
  const lib = path.join(dir, "Library.lua");
  const src = readFileSync(lib, "utf8");
  if (!src.includes(from)) throw new Error(`${id}: text not found`);
  writeFileSync(lib, src.replace(from, to));
  let out: string;
  try {
    out = execFileSync(process.execPath, [path.join(here, "run.mts"), dir], { encoding: "utf8" });
  } catch (err) {
    out = String((err as { stdout?: string }).stdout ?? err);
  }
  const failed = out.split(/\r?\n/).filter((l) => l.startsWith("FAIL ") || l.startsWith("LUA ERROR"));
  if (failed.length > 0) caught++;
  console.log(`${id}: ${failed.length > 0 ? "CAUGHT" : "SURVIVED"}`);
  for (const l of failed) console.log(`    ${l}`);
}
console.log(`${caught} of ${MUTANTS.length} mutants caught`);

  if (!src.includes(from)) throw new Error(`${id}: text not found`);
  let mutated = src.replace(from, to);
  // E1 closes the whenIdle call it opened: "});" of the whenIdle becomes "})();" for the IIFE.
  if (id.startsWith("E1")) mutated = mutated.replace('      return written(out, { rating: args.rating });\r\n    });', "      return written(out, { rating: args.rating });\r\n    })();").replace('      return written(out, { rating: args.rating });\n    });', "      return written(out, { rating: args.rating });\n    })();");
  writeFileSync(file, mutated);
  let out = "";
  try {
    out = execSync("npx vitest run tests/catalog-tools.test.ts", { cwd: path.join(repo, "engine"), encoding: "utf8", stdio: "pipe" });
  } catch (err) {
    out = String((err as { stdout?: string }).stdout ?? "") + String((err as { stderr?: string }).stderr ?? "");
  } finally {
    writeFileSync(file, src);
  }
  const failed = [...new Set(out.split(/\r?\n/).filter((l) => /FAIL .*>/.test(l)).map((l) => l.trim()))];
  if (failed.length > 0) caught++;
  console.log(`${id}: ${failed.length > 0 ? "CAUGHT" : "SURVIVED"}`);
  for (const l of failed) console.log(`    ${l}`);
}
console.log(`${caught} of ${MUTANTS.length} mutants caught`);
