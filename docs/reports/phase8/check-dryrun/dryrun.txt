// SCRATCH (not committed): dry run of the built `phase8-check-cli.js` against the simulated plugin on
// OS-chosen ports with its own token file, so Jim's Lightroom is not touched. The real
// `engine\dist\mcp\main.js` plays Claude Desktop in Part 5 over stdio MCP, kept open as Claude Desktop
// keeps its engine, so the check takes the bridge back by the engine's real 60 s idle release.
// LRC_AVG_LOG_DIR, LRC_AVG_P8_OUT, LRC_AVG_PLUGIN_LOG, LRC_AVG_PRESET_DIR and LOCALAPPDATA point at a
// scratch folder. "Jim" acts on the CLI's words as tests\helpers\phase8-harness.ts does.

import { spawn } from "node:child_process";
import { copyFileSync, mkdirSync, mkdtempSync, readdirSync, readFileSync, unlinkSync, writeFileSync } from "node:fs";
import os from "node:os";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { Client } from "@modelcontextprotocol/sdk/client/index.js";
import { StdioClientTransport } from "@modelcontextprotocol/sdk/client/stdio.js";
import { expect, it } from "vitest";
import { parseXml, presetDescription } from "../src/presets/index.js";
import { defaultPreviewDir } from "../src/preview/index.js";
import { FakePlugin } from "./helpers/fake-plugin.js";
import { LightroomSim, nefDump } from "./helpers/lightroom-sim.js";
import { renderedDumps } from "./helpers/lightroom-sim-rendered.js";

const repo = fileURLToPath(new URL("../../", import.meta.url));
const FIXTURES = fileURLToPath(new URL("./fixtures/presets/", import.meta.url));

it("dry run of npm run phase8:check", { timeout: 1800000 }, async () => {
  const scratch = mkdtempSync(path.join(os.tmpdir(), "lrc-avg-p8dry-"));
  for (const d of ["logs", "out", "localappdata", "presets"]) mkdirSync(path.join(scratch, d));
  const presetDir = path.join(scratch, "presets");
  copyFileSync(path.join(FIXTURES, "reference-rendered.lrc15.xmp"), path.join(presetDir, "AVG preset reference rendered.xmp"));
  copyFileSync(path.join(FIXTURES, "reference-rendered-mono.lrc15.xmp"), path.join(presetDir, "AVG preset reference rendered mono.xmp"));
  const bridgeLog = path.join(scratch, "bridge.log");
  writeFileSync(bridgeLog, "");
  const plugin = await FakePlugin.start();
  plugin.token = "dry-token-p8";
  const tokenFile = path.join(scratch, "dry_token");
  writeFileSync(tokenFile, plugin.token);
  const lr = new LightroomSim(defaultPreviewDir());
  lr.renderModel = "tonal";
  const lib = (uuid: string, local_id: number, filename: string, file_format: string, settings: Record<string, unknown>, copy_name?: string) =>
    lr.library.photos.push({ uuid, local_id, filename, rating: 0, keywords: [], day: "2004-09-25", gps: null, file_format, settings: structuredClone(settings), ...(copy_name ? { copy_name } : {}) });
  lib("SIM-NEF2", 11, "_DSC0103.NEF", "RAW", nefDump.settings);
  lib("SIM-JPG", 12, "DSC_0031.JPG", "JPG", renderedDumps["15.4"].settings);
  lib("SIM-JPG-11", 13, "DSC_0031.JPG", "JPG", renderedDumps["11.0"].settings);
  lib("SIM-JPG-C1", 14, "DSC_0031.JPG", "JPG", renderedDumps["15.4"].settings, "Copy 1");
  lib("SIM-TIF", 15, "20260907-_OZ80099-Edit.tif", "TIFF", renderedDumps["15.4"].settings);
  lr.missing.add("SIM-TIF");
  lr.library.collections.push({ local_id: 503, name: "fixtures", smart: false, photos: [lr.uuid, "SIM-NEF2", "SIM-JPG", "SIM-JPG-11", "SIM-JPG-C1", "SIM-TIF"] });
  lr.install(plugin);
  const env = {
    ...process.env,
    LRC_AVG_COMMAND_PORT: String(plugin.commandPort),
    LRC_AVG_EVENT_PORT: String(plugin.eventPort),
    LRC_AVG_TOKEN_FILE: tokenFile,
    LRC_AVG_LOG_DIR: path.join(scratch, "logs"),
    LRC_AVG_P8_OUT: path.join(scratch, "out"),
    LRC_AVG_PLUGIN_LOG: bridgeLog,
    LRC_AVG_PRESET_DIR: presetDir,
    LOCALAPPDATA: path.join(scratch, "localappdata"),
  };

  // "Claude Desktop": the real engine over stdio MCP.
  let desk: Client | null = null;
  const call = async (name: string, args: Record<string, unknown>): Promise<Record<string, unknown>> => {
    if (!desk) {
      desk = new Client({ name: "dry-desktop", version: "0" });
      await desk.connect(new StdioClientTransport({ command: process.execPath, args: [path.join(repo, "engine/dist/mcp/main.js")], env: env as Record<string, string>, stderr: "pipe" }));
    }
    const r = (await desk.callTool({ name, arguments: args }, undefined, { timeout: 180000 })) as { content: Array<{ type: string; text?: string }> };
    return JSON.parse(r.content.find((x) => x.type === "text")?.text ?? "{}") as Record<string, unknown>;
  };
  const chat = async (): Promise<string> => {
    const begin = await call("lr_begin_session", { intent_id: "portrait_natural_light", return_image: "none" });
    const sid = String(begin["session_id"]);
    await call("lr_step", { session_id: sid, settings: { vibrance: 3 }, rationale: "dry run", return_image: "none" });
    const end = await call("lr_end_session", { session_id: sid, outcome: "accept" });
    return `chat: begin ${String(begin["ok"])} on the ${String(begin["pipeline"])} pipeline, end ${String(end["outcome"])}`;
  };
  const clickPreset = (name: string): string => {
    const file = readdirSync(presetDir, { recursive: true, encoding: "utf8" }).find((f) => path.basename(f) === `${name}.xmp`);
    if (!file) return `preset ${name}: no file`;
    const target = lr.settingsOf(lr.selected);
    let n = 0;
    for (const [attr, raw] of presetDescription(parseXml(readFileSync(path.join(presetDir, file), "utf8"))).attrs) {
      const key = attr.replace(/^crs:/, "");
      if (!attr.startsWith("crs:") || !(key in target) || (key === "CameraProfile" && raw.startsWith("Default "))) continue;
      const was = target[key];
      target[key] = typeof was === "number" ? Number(raw) : typeof was === "boolean" ? raw === "True" : raw;
      n++;
    }
    return `preset ${name} clicked on ${lr.selected}: ${n} keys`;
  };

  const notes: string[] = [];
  const cli = spawn(process.execPath, [path.join(repo, "engine/dist/devtools/phase8-check-cli.js")], { env, stdio: ["pipe", "pipe", "pipe"] });
  let out = "";
  let tail = "";
  cli.stderr.on("data", (d) => (out += String(d)));
  const enter = (s = ""): void => void cli.stdin.write(`${s}\n`);
  let pending = Promise.resolve();
  const prompt = async (text: string): Promise<boolean> => {
    if (/Type y or n, then Enter:\s*$/.test(text)) enter("y");
    else if (/3\. Then press Enter here\.\s*$/.test(text)) {
      plugin.dropEventClient(); // Lightroom quit and started again
      enter();
    } else if (/click "(AVG P8check [^"]+)" once\. Then press Enter here\.\s*$/.test(text)) {
      notes.push(clickPreset(/click "(AVG P8check [^"]+)" once/.exec(text)?.[1] as string));
      enter();
    } else if (/6\. Come back to this window and press Enter\.\s*$/.test(text)) {
      notes.push(await chat());
      enter();
    } else if (/^ {2}3\. Press Enter here\.\s*$/.test(text)) {
      for (const f of readdirSync(presetDir)) if (f.endsWith(".xmp")) unlinkSync(path.join(presetDir, f));
      enter();
    } else if (/[Pp]ress Enter[^\n]*\.\s*$/.test(text)) enter();
    else return false;
    return true;
  };
  cli.stdout.on("data", (d) => {
    const text = String(d);
    out += text;
    pending = pending.then(async () => {
      const lines = (tail + text).split("\n");
      tail = lines.pop() ?? "";
      if (tail && (await prompt(tail))) tail = "";
    });
  });
  const code = await new Promise((r) => cli.on("close", r));
  await (desk as Client | null)?.close();
  await plugin.close();
  const transcript = out.split(os.homedir()).join("%USERPROFILE%");
  writeFileSync(path.join(repo, "logs", "scratch-p8-dryrun-output.txt"), `${transcript}\n[driver] ${notes.join("\n[driver] ")}\n[driver] exit code ${String(code)}; History steps ${lr.history.length}; exports ${lr.exports}; scratch ${scratch.split(os.homedir()).join("%USERPROFILE%")}\n`);
  expect(code).toBe(0);
});
