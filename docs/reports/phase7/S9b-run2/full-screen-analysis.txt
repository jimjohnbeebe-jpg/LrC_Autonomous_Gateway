// S9b run 2: the HUD's log from Lightroom's restore to the end, beside the harness's last S9-8 episodes.
import { readFileSync } from "node:fs";
const dir = "docs/reports/phase7/S9b-run2/";
const run = JSON.parse(readFileSync(dir + "s9_s9b-2026-10-05T12-36-43-456Z.json", "utf8"));
const t0 = Date.parse(run.started);
const log = readFileSync(dir + "hud_3912_1791203830873.jsonl", "utf8").trim().split("\n").map((l) => JSON.parse(l));
console.log("== A. Harness S9-8 episodes 10-13 (minimise, restore, and the two after it)");
run.s9_8.episodes.slice(9).forEach((e: any, i: number) =>
  console.log(i + 10, e.kind, "ms=" + (e.ms?.toFixed(1) ?? null), "detached_ms=" + e.detached_ms,
    "lr=" + (e.lr_rect ? Object.values(e.lr_rect).join(",") : "-"), "hud=" + Object.values(e.hud_rect).join(","), "visible=" + e.hud_visible));
console.log("== B. HUD log from 349 s (seconds from the run start); 'follow' = a location event from Lightroom's process");
for (const e of log) {
  const s = (e.t - t0) / 1000;
  if (s < 349) continue;
  console.log(s.toFixed(3), e.ev, e.reason ?? e.kind ?? e.event ?? "", e.got ? JSON.stringify(e.got) : "", e.fg_pid ?? "");
}
