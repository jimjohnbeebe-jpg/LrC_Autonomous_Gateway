===== fake-lr.lua =====
-- A stubbed Lightroom SDK for smoke-running plugin Lua under fengari (no Lightroom). It imitates only
-- what the LrC-AVG plugin calls, with the behaviour observed in the spikes (S6, S7) where there is
-- one, and records what the plugin did so the scenarios can check it. It also enforces two plugin
-- rules: yielding catalog queries never run inside a read gate (rule 03), and getRawMetadata /
-- getFormattedMetadata / getDevelopSettings only run inside one.

F = { clock = 1000, tasks = {}, files = {}, errors = {}, yields = 0, inRead = 0, inWrite = 0,
      calls = {}, binds = {}, uuidN = 0, nextLocalId = 5000 }

local function record(what) F.calls[#F.calls + 1] = what end

-- In-memory files (fengari has no io.open).
io = io or {}
io.open = function(path, mode)
    mode = mode or "r"
    if mode:sub(1, 1) == "r" then
        if F.files[path] == nil then return nil, path .. ": No such file" end
        local text = F.files[path]
        return { read = function() return text end, close = function() end }
    end
    if mode:sub(1, 1) == "w" then F.files[path] = "" end
    if F.files[path] == nil then F.files[path] = "" end
    return {
        write = function(self, ...) for _, s in ipairs({ ... }) do F.files[path] = F.files[path] .. tostring(s) end end,
        close = function() end,
    }
end

-- A tiny cooperative scheduler: every task is a coroutine; sleep and yield hand control back.
function F.spawn(name, fn)
    F.tasks[#F.tasks + 1] = { name = name, co = coroutine.create(fn) }
end
function F.run(rounds)
    for _ = 1, rounds or 1 do
        local live = {}
        for _, t in ipairs(F.tasks) do
            if coroutine.status(t.co) == "suspended" then
                local ok, err = coroutine.resume(t.co)
                if not ok then F.errors[#F.errors + 1] = t.name .. ": " .. tostring(err) end
            end
            if coroutine.status(t.co) ~= "dead" then live[#live + 1] = t end
        end
        F.tasks = live
    end
end
-- Run fn in a task until it ends; returns what it returned.
function F.runTask(fn)
    local out
    F.spawn("runTask", function() out = table.pack(fn()) end)
    for _ = 1, 1000 do
        if #F.tasks == 0 then break end
        F.run(1)
    end
    if not out then return nil end
    return table.unpack(out, 1, out.n)
end

local M = {}

M.LrTasks = {
    pcall = pcall,
    yield = function() F.yields = F.yields + 1; coroutine.yield() end,
    sleep = function(s) F.clock = F.clock + (s or 0); coroutine.yield() end,
    startAsyncTask = function(fn) F.spawn("async", fn) end,
}
M.LrDate = {
    currentTime = function() return F.clock end,
    timeToUserFormat = function(t, fmt) return "2026-09-27 12:00:00" end,
}
M.LrUUID = { generateUUID = function() F.uuidN = F.uuidN + 1; return string.format("UUID-%04d-AAAA-BBBB", F.uuidN) end }
M.LrPathUtils = {
    child = function(a, b) return a .. "\\" .. b end,
    parent = function(p) return (p:gsub("\\[^\\]*$", "")) end,
    getStandardFilePath = function(which) return which == "home" and "C:\\Users\\U" or "C:\\T" end,
}
M.LrFileUtils = {
    createAllDirectories = function() end,
    fileAttributes = function() return nil end,
    delete = function() end,
    files = function() return function() return nil end end,
}
M.LrPrefs = { prefsForPlugin = function() return {} end }

-- Sockets: each bind returns a socket object; the scenario drives its callbacks.
M.LrSocket = {
    bind = function(spec)
        local sock = { spec = spec, sent = {}, closed = false, reconnects = 0 }
        function sock:send(line) self.sent[#self.sent + 1] = line end
        function sock:close() self.closed = true end
        function sock:reconnect() self.reconnects = self.reconnects + 1 end
        F.binds[#F.binds + 1] = sock
        F.last = F.last or {}
        F.last[spec.mode] = sock
        return sock
    end,
}
M.LrFunctionContext = {
    postAsyncTaskWithContext = function(name, fn)
        local ctx = { cleanups = {} }
        function ctx:addCleanupHandler(h) self.cleanups[#self.cleanups + 1] = h end
        F.contexts = F.contexts or {}
        F.contexts[#F.contexts + 1] = ctx
        F.spawn(name, function() fn(ctx) end)
    end,
}

-- Photos and the catalog.
F.photos, F.byUuid = {}, {}
function F.newPhoto(t)
    local p = { localIdentifier = t.local_id }
    p._t = t
    function p:getRawMetadata(key)
        assert(F.inRead > 0, "getRawMetadata outside a read gate")
        if key == "uuid" then return t.uuid end
        if key == "isVirtualCopy" then return t.is_vc == true end
        if key == "masterPhoto" then return t.master or p end
        return t[key]
    end
    function p:getFormattedMetadata(key)
        assert(F.inRead > 0, "getFormattedMetadata outside a read gate")
        if key == "copyName" then return t.copy_name end
        if key == "fileName" then return t.file_name end
        return nil
    end
    function p:getDevelopSettings()
        assert(F.inRead > 0, "getDevelopSettings outside a read gate")
        return { Exposure2012 = 0 }
    end
    F.photos[t.local_id] = p
    F.byUuid[t.uuid] = p
    return p
end

local function noQueryInGate(what)
    assert(F.inRead == 0, what .. " inside a read gate (rule 03: deadlocks on Windows)")
    -- With F.yieldQueries, a catalog query hands control to other tasks, as a yielding query would.
    if F.yieldQueries and coroutine.isyieldable() then coroutine.yield() end
end

local function overlap(name, copyYields, delay)
    F.world()

-- A lock whose holder never finishes (Greptile, PR #33 round 2). `start` runs a command as its own
-- task after `delay` rounds; F.kill stops a task for good, as a Reload Plug-in may.
local LOG_FILE = "C:\\T\\LrC-AVG\\bridge.log"
local function start(results, label, delay, fn)
    F.spawn(label, function()
        for _ = 1, delay or 0 do coroutine.yield() end
        results[label] = table.pack(fn())
        results[label .. "_at"] = F.clock
    end)
end
local function runAll()
    for _ = 1, 5000 do
        if #F.tasks == 0 then break end
        F.run(1)
    end
end
function F.kill(label)
    local keep = {}
    for _, t in ipairs(F.tasks) do if t.name ~= label then keep[#keep + 1] = t end end
    F.tasks = keep
end
local function takeovers()
    local lines = {}
    for line in (F.files[LOG_FILE] or ""):gmatch("[^\n]+") do
        if line:find("takes over", 1, true) then lines[#lines + 1] = line:gsub("^%S+ %S+ ", "") end
    end
    return lines
end

    local p = r[label]
    return p and { result = p[1], error = p[2], at = r[label .. "_at"] } or { unfinished = true }

"ok" : c.select.error?.code}`; };
