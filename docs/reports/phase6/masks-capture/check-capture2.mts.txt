async function ask(q: string): Promise<string> {
  let a: string | null = null;
  for (let i = 0; i < 3 && a !== "y" && a !== "n" && !aborting; i++) {
    try {
      a = (await rl.question(`${q} (y/n) `, { signal: AbortSignal.any([AbortSignal.timeout(600_000), abortCtl.signal]) })).trim().toLowerCase();
    } catch {
      break; // no answer in 10 minutes, aborted, or stdin closed
    }
  }

  result.answers.push({ q, answer, at: at() });
  transcript.push(`  ? ${q} -> ${answer}`);
  if (answer !== "y" && answer !== "aborted") result.failures.push(`answered ${answer}: ${q}`);
  save();
  return answer;
}

const isObj = (v: J) => v !== null && typeof v === "object" && !Array.isArray(v);
const blank = (v: J) => v === undefined || (Array.isArray(v) && v.length === 0) || (isObj(v) && Object.keys(v).length === 0);
const short = (v: J) => (v === undefined ? "(absent)" : JSON.stringify(v).slice(0, 160));
function diff(a: J, b: J, where = "", out: Array<Record<string, J>> = []): Array<Record<string, J>> {
  if (out.length >= 200 || (blank(a) && blank(b))) return out;
  if (Array.isArray(a) && Array.isArray(b)) {
    if (a.length !== b.length) out.push({ path: `${where}.length`, before: a.length, after: b.length });
    for (let i = 0; i < Math.min(a.length, b.length); i++) diff(a[i], b[i], `${where}[${i}]`, out);
  } else if (isObj(a) && isObj(b)) {
    for (const k of new Set([...Object.keys(a), ...Object.keys(b)])) diff(a[k], b[k], where ? `${where}.${k}` : k, out);
  } else if (a !== b) out.push({ path: where || "(root)", before: short(a), after: short(b) });
  return out;
}
function newIdLike(s: string): string {
  const upper = /[A-F]/.test(s) && !/[a-f]/.test(s);
  return s.replace(/[0-9a-fA-F]/g, () => {
    const c = "0123456789abcdef".charAt(randomInt(16));
    return upper ? c.toUpperCase() : c;
  });
}
/** A copy without fields whose names contain "Digest", with new values for string fields named *ID. */
function freshCopy(v: J): J {
  if (Array.isArray(v)) return v.map(freshCopy);
  if (!isObj(v)) return v;
  const out: J = {};
  for (const [k, x] of Object.entries(v)) {
    if (k.includes("Digest")) continue;
    out[k] = /ID$/.test(k) && typeof x === "string" && x !== "" ? newIdLike(x) : freshCopy(x);
  }
  return out;
}

async function run(name: string, fn: () => Promise<string | void>): Promise<boolean> {
  if (aborting) return false;
  say(`--- ${name}`);
  try {
    const skipped = await fn();
    if (skipped) (step(`${name}_skipped`, { reason: skipped }), say(`skipped: ${skipped}`));
    return true;
  } catch (err) {
    step(`${name}_error`, { error: errBody(err), aborting });
    if (!aborting) check(false, `${name} ran`, errBody(err));
    return false;
  }
}

/** After a probe that failed: read every second until the settings held still for 5 s, at most 90 s. */
async function waitStable(): Promise<Record<string, J>> {
  const t0 = Date.now();
  let last = JSON.stringify(await read());
  let since = Date.now();
  while (Date.now() - since < 5_000 && Date.now() - t0 < 90_000) {
    await sleep(1000);
    const now = JSON.stringify(await read());
    if (now !== last) {
      last = now;
      since = Date.now();
    }
  }
  return { stable: Date.now() - since >= 5_000, waited_ms: Date.now() - t0 };
}
/** A probe, with the stable-wait after a failure; the answer, or { error }. None is sent after Ctrl+C. */
async function probe(name: string, payload: J, timeoutMs: number): Promise<J> {
  if (aborting) return { error: "not sent: aborting", skipped: true };
  const out = await req(name, { target_uuid: UUID, ...payload }, timeoutMs).catch((err) => ({ error: errBody(err) }));
  if (out.error) out.settled = await waitStable();
  return out;
}

/** Capture 1's linear entry, the shape the calibration correction is copied from. */
function template(): J {
  for (const file of TEMPLATES) {
    if (!existsSync(file)) continue;
    const entry = masks(JSON.parse(readFileSync(file, "utf8"))).find((e) => (e.CorrectionMasks ?? []).some((m: J) => m.What === "Mask/Gradient"));
    if (entry) return { file, entry };
  }
  throw new Error(`no linear gradient entry in capture 1's dump (looked in ${TEMPLATES.join("; ")})`);
}

/**
 * Round r (0-2): writes every field at its r-th value, reads the entry back, probes it. Each round's
 * entry is a fresh copy with new ids (CorrectionID, MaskID, sync ids), so the probe cannot read the
 * previous round's mask, which it had already selected.
 */
async function round(r: number): Promise<string | void> {
  if (r === 0) {
    const t = template();
    st.template = t.entry;
    step("2_template", { file: t.file });
  }
  if (!st.template) return "round 1 did not run";
  const entry = { ...freshCopy(st.template), CorrectionName: NAME, ...Object.fromEntries(Object.entries(FIELDS).map(([f, v]) => [f, v[r]])) };
  const rb = await write({ [KEY]: [...masks(st.before), entry] }, `AVG calibrate write ${r + 1}`);
  const back = masks(rb).find((e) => e.CorrectionID === entry.CorrectionID);
  saveRaw(`round${r + 1}_entry`, back ?? null);
  check(back !== undefined, `round ${r + 1}: the "${NAME}" correction is in the table after the write`);
  if (!back) return "the correction was not taken";
  const stored = Object.fromEntries([...Object.keys(FIELDS), ...RECORDED].map((f) => [f, back[f]]));
  const changed = Object.keys(FIELDS).filter((f) => stored[f] !== FIELDS[f]![r]);
  check(changed.length === 0, `round ${r + 1}: every slider field reads back as written${changed.length ? ` (not: ${changed.join(", ")})` : ""}`, changed.map((f) => ({ f, wrote: FIELDS[f]![r], read: stored[f] })));
  if (r === 0) say("The plugin now selects the mask in Develop > Masking and reads its sliders. Do not click in Lightroom until the next question.");
  const p = await probe("probe_masks_calibrate", { mask_id: entry.CorrectionID, mask_name: NAME, names: NAMES }, 75_000);
  if (p.skipped) return "aborted before the probe";
  saveRaw(`round${r + 1}_probe`, p);
  const selectedIt = !p.error && p.selected === entry.CorrectionID;
  // Values read with another mask selected would mislead: the round's panel values count only when it was this round's mask.
  const panel = Object.fromEntries(NAMES.map((n) => [n, stepResult(p, `getValue_${n}`)]).map(([n, s]) => [n, selectedIt && s?.ok ? s.result : { error: s?.error ?? "not read" }]));
  const ranges = Object.fromEntries(NAMES.map((n) => [n, stepResult(p, `getRange_${n}`)?.result ?? null]));
  (st.rounds ??= [])[r] = { correction_id: entry.CorrectionID, stored, panel, ranges, selected: p.selected ?? null };
  step(`round${r + 1}`, { correction_id: entry.CorrectionID, error: p.error ?? null, settled: p.settled ?? null, stopped: p.stopped ?? null, selected: p.selected ?? null, tool_end: stepResult(p, "getSelectedTool_end")?.result ?? null, stored, panel });
  check(selectedIt, `round ${r + 1}: the probe selected this round's "${NAME}" (${p.error ?? p.stopped ?? `selected ${p.selected ?? "nothing"}`})`);
}

/** Per field: the name that answered, stored -> panel per round, the fit and its flags. */
function fit(field: string): Record<string, J> {
  const rounds: J[] = st.rounds ?? [];
  const stored = rounds.map((r) => r?.stored?.[field]);
  const tried = candidates(field).map((name) => ({ name, panel: rounds.map((r) => r?.panel?.[name]) }));
  // The first name whose answer follows the table (rounds 1 and 2 differ), else the first that answered at all.
  const numeric = tried.filter((t) => t.panel.slice(0, 2).every((v) => typeof v === "number"));
  const answered = numeric.find((t) => t.panel[0] !== t.panel[1]) ?? numeric[0];
  if (!FIELDS[field]) return { field, name: answered?.name ?? null, tried, stored, panel: answered?.panel ?? null, flags: ["recorded, not written"] };
  const flags: string[] = [];
  if (stored.some((s, i) => s !== undefined && s !== FIELDS[field]![i])) flags.push("stored differs from written");
  if (rounds.filter(Boolean).length < 2) return { field, tried, stored, flags: [...flags, "fewer than 2 rounds"] };
  if (!answered) return { field, tried, stored, flags: [...flags, "no DC answer"] };
  const [p1, p2, p3] = answered.panel as number[];
  const [s1, s2, s3] = stored as number[];
  if (p1 === p2) return { field, name: answered.name, tried, stored, panel: answered.panel, flags: [...flags, "DC constant (did not follow the table)"] };
  const scale = (p2! - p1!) / (s2! - s1!);
  const offset = p1! - scale * s1!;
  const predicted = typeof s3 === "number" ? scale * s3 + offset : null;
  if (typeof p3 !== "number" || predicted === null) flags.push("no round 3");
  else if (Math.abs(p3 - predicted) > Math.max(1e-6, 0.01 * Math.abs(p2! - p1!))) flags.push(`non-linear (round 3: ${fmt(p3)}, line gives ${fmt(predicted)})`);
  return { field, name: answered.name, tried, stored, panel: answered.panel, scale, offset, range: rounds[0]?.ranges?.[answered.name] ?? null, flags };
}

async function s6(): Promise<void> {
  const fields = Object.keys(FIELDS).map(fit);
  const recorded = RECORDED.map(fit);
  writeFileSync(path.join(outDir, "calibration.json"), JSON.stringify({ at: at(), uuid: UUID, values: FIELDS, names: NAMES, fields, recorded, question: st.question ?? null, rounds: st.rounds ?? [] }, null, 2));
  say("Results (stored -> panel per round; panel = scale * stored + offset):");
  say(`${"field".padEnd(28)}${"DC name".padEnd(28)}${"stored -> panel".padEnd(44)}${"scale".padEnd(10)}${"offset".padEnd(8)}flags`);
  for (const f of [...fields, ...recorded]) {
    const pairs = (f.stored as J[]).map((s, i) => `${fmt(s)}->${fmt(typeof f.panel?.[i] === "number" ? f.panel[i] : null)}`).join(" ");
    say(`${f.field.padEnd(28)}${String(f.name ?? "-").padEnd(28)}${pairs.padEnd(44)}${fmt(f.scale).padEnd(10)}${fmt(f.offset).padEnd(8)}${f.flags.join("; ")}`);
  }
  const clean = fields.filter((f) => f.flags.length === 0).length;
  step("6_results", { clean, of: fields.length });
  check(clean === fields.length, `every slider field has a clean linear scale (${clean} of ${fields.length}; calibration.json)`);
}

/**
 * A subtype's mask: only when its mask showed (`wait_mask_<sub>` reached) and exactly one id is new
 * since just before its create (`new_masks_<sub>`), not claimed by an earlier subtype and not in the
 * table before the probe. Otherwise null and why: a subtype that made no mask must never be credited
 * with the mask still selected from the one before.
 */
function claim(p: J, sub: string, claimed: Set<string>, old: Set<string>): { id: string | null; why: string | null } {
  const shown = stepResult(p, `wait_mask_${sub}`)?.result;
  if (shown?.reached !== true) return { id: null, why: "no new mask showed" };
  const fresh: string[] = (stepResult(p, `new_masks_${sub}`)?.result ?? []).filter((id: string) => !claimed.has(id) && !old.has(id));
  if (fresh.length !== 1) return { id: null, why: `${fresh.length} unclaimed new ids` };
  return { id: fresh[0]!, why: null };
}

async function s8(): Promise<string | void> {
  say("The plugin now creates AI masks in Develop by itself (up to a minute). Do not click in Lightroom.");
  const p = await probe("probe_masks_create", { subtypes: AI.subtypes, wait_seconds: AI.wait }, 90_000);
  if (p.skipped) return "aborted before the probe";
  saveRaw("8_create_probe", p);
  await sleep(3000); // let a mask still computing finish [inference: capture 1's DC masks took 0.9-2.5 s]
  const s = await read();
  const old = new Set<string>(masks(st.before).map((e) => e.CorrectionID));
  const fresh = masks(s).filter((e) => !old.has(e.CorrectionID));
  const claimed = new Set<string>();
  const templates: Record<string, J> = {};
  for (const sub of AI.subtypes) {
    const { id, why } = claim(p, sub, claimed, old);
    if (id) claimed.add(id);
    templates[sub] = { dc_id: id, why, shown: stepResult(p, `wait_mask_${sub}`)?.result ?? null, tool: stepResult(p, `getSelectedTool_after_${sub}`)?.result ?? null, entry: fresh.find((e) => e.CorrectionID === id) ?? null };
  }
  writeFileSync(path.join(outDir, "templates.json"), JSON.stringify({ at: at(), templates, unmatched: fresh.filter((e) => !claimed.has(e.CorrectionID)), probe: p }, null, 2));
  saveRaw("8_after_create", s);
  step("8_ai_masks", { error: p.error ?? null, stopped: p.stopped ?? null, tool_end: stepResult(p, "getSelectedTool_end")?.result ?? null, ...Object.fromEntries(Object.entries(templates).map(([k, t]) => [k, { dc_id: t.dc_id, why: t.why, shown: t.shown, tool: t.tool, in_table: t.entry !== null }])) });
  for (const sub of AI.subtypes) check(templates[sub].entry !== null, `createNewMask("aiSelection", "${sub}") made its own table entry (templates.json)${templates[sub].why ? `: ${templates[sub].why}` : ""}`, templates[sub].shown);
}

const STEPS: Array<[string, () => Promise<string | void>]> = [
  ["2_round1", () => round(0)], ["3_question", s5], ["4_round2", () => round(1)], ["5_round3", () => round(2)],
  ["6_results", async () => s6()], ["7_snapshot", s7], ["8_ai_masks", s8],
];
try {
  if ((await run("0_preconditions", s0)) && (await run("1_photo", s1)) && st.snapshotId && st.before) {
    for (const [name, fn] of STEPS) {
      if (process.env["CHECK_SIGINT_AT"] === name) rl.emit("SIGINT"); // dry runs only: a Ctrl+C before this step
      if (aborting) break;
      await run(name, fn);
    }
  }
} finally {
  // 9. The put-back, whatever happened (also after Ctrl+C). The results table first, from whatever rounds ran.
  if (st.rounds && !result.steps["6_results"]) await s6().catch((err) => step("6_results_error", { error: errBody(err) }));
  const manual = `wait 1 minute, then in Lightroom's Develop module open the Snapshots panel (left side) and click "${st.snapshotName ?? "AVG calibrate before ..."}"`;
  try {
    if (st.snapshotId && st.before) {
      say("--- 9_put_back");
      const d = await restore("9_put_back");
      // A mask still computing could change the settings after the snapshot: read them again 5 s later.
      await sleep(5000);
      const late = diff(st.before, await read());
      step("9_put_back_late", { differing: late });
      const all = [...d, ...late];
      check(all.length === 0, all.length === 0 ? "PUT BACK: YES (every setting as before the check, and still 5 s later)" : `PUT BACK: NO (${d.length} differing paths, ${late.length} 5 s later): ${manual}`, { at_once: d, late });
      if (!aborting) {
        say("NOW: look at the photo in Lightroom, and at the Masks panel (if the Masks panel is not open, press Shift+W).");
        await ask("Does the photo look as it did before the check, with none of the check's masks left?");
      }
    }
  } catch (err) {
    check(false, `PUT BACK: NO, it did not run (${errBody(err)}): ${manual}`, errBody(err));
  }
  result.bridge = client.stats;
  result.ended = at();
  save();
  client.stop();
  await lock.lock.release();
  rl.close();
  const failures: string[] = result.failures;
  say("");
  say(failures.length === 0 ? "masks capture 2: WORKED" : `masks capture 2: DONE, ${failures.length} NO / FAILED lines (each one is a finding for the report)`);
  for (const f of failures) say(`  - ${f}`);
  say(`Results: ${path.join(outDir, "check.json")}, calibration.json, templates.json`);
  save();
  process.exit(failures.length === 0 ? 0 : 1);
}
