local function fileHandle(path, mode)
    local h = { path = path, mode = mode, buf = {} }
    if mode:find("a") then h.buf[1] = FAKE.files[path] or "" end
    function h:write(...)
        for _, s in ipairs({ ... }) do table.insert(self.buf, tostring(s)) end
        return self
    end

    function h:lines()
        local t = FAKE.files[self.path] or ""
        return t:gmatch("([^\n]*)\n?")
    end
    function h:close()
        if self.mode:find("[wa]") then FAKE.files[self.path] = table.concat(self.buf) end
        return true
    end

end

function FAKE.run(untilTime)
    while true do
        for i = #FAKE.tasks, 1, -1 do
            if coroutine.status(FAKE.tasks[i].co) == "dead" then table.remove(FAKE.tasks, i) end
        end
        local best
        for i, t in ipairs(FAKE.tasks) do
            local b = best and FAKE.tasks[best]
            if not b or t.wake < b.wake or (t.wake == b.wake and t.seq < b.seq) then best = i end
        end
        if not best or FAKE.tasks[best].wake > untilTime then
            if untilTime > FAKE.now then FAKE.now = untilTime end
            return
        end
        local t = FAKE.tasks[best]
        if t.wake > FAKE.now then FAKE.now = t.wake end
        local ok, err = coroutine.resume(t.co)
        if not ok then flog("TASK ERROR in " .. t.name .. ": " .. tostring(err)); error("task error in " .. t.name .. ": " .. tostring(err)) end
    end
end

    coroutine.yield()

// Runs main.lua under fengari for one scenario; writes the fake's files to out/<scenario>/ and
// prints the checks. Usage: node run-smoke.cjs <pluginDir> <scenario>
const fs = require("fs");
const path = require("path");
const { lua, lauxlib, lualib, to_luastring } = require("fengari");

const [pluginDir, scenario] = process.argv.slice(2);
const here = __dirname.replaceAll("\\", "/");
const L = lauxlib.luaL_newstate();
lualib.luaL_openlibs(L);
const prelude = `PLUGIN_DIR = ${JSON.stringify(pluginDir.replaceAll("\\", "/"))}; FAKE_DIR = ${JSON.stringify(here)}; SCENARIO = ${JSON.stringify(scenario)}`;
const code = prelude + "\n" + fs.readFileSync(path.join(__dirname, "main.lua"), "utf8");
let out = "";
const origLog = console.log;
const origWrite = process.stdout.write.bind(process.stdout);
process.stdout.write = (s) => { out += s; return true; };
let status = lauxlib.luaL_loadbuffer(L, to_luastring(code), null, to_luastring("main"));
if (status === lua.LUA_OK) status = lua.lua_pcall(L, 0, 0, 0);
process.stdout.write = origWrite;
if (status !== lua.LUA_OK) {
  origLog(out);
  origLog("LUA ERROR: " + lua.lua_tojsstring(L, -1));
  process.exit(2);
}
const outDir = path.join(__dirname, "out", scenario);
fs.rmSync(outDir, { recursive: true, force: true });
fs.mkdirSync(outDir, { recursive: true });
const parts = out.split(/^===/m);
const report = [];
for (const p of parts) {
  if (p.startsWith("FILE ")) {
    const nl = p.indexOf("\n");
    fs.writeFileSync(path.join(outDir, p.slice(5, nl).trim()), p.slice(nl + 1).replace(/\n$/, ""));
  } else report.push(p.startsWith("LOG") || p.startsWith("RESULT") ? "===" + p : p);
}
// late_close also leaves the live loop-view file half-written, as a read during a rewrite would
// find it, for the collector's dry run.
if (scenario === "late_close") {
  const live = path.join(outDir, "s8_loop_view.json");
  const text = fs.readFileSync(live, "utf8");
  fs.writeFileSync(live, text.slice(0, Math.floor(text.length / 2)));
  report.push("(run-smoke: s8_loop_view.json cut to half its length for the collector dry run)\n");
}
origLog(report.join(""));
// A failed check fails the run, so the transcript builder can tell (Greptile, PR #40 round 2).
if (!/^===RESULT ALL PASS/m.test(out)) process.exit(1);

// Parse every .lua under the given folders as Lua 5.1 (syntax only).
const fs = require("fs");
const path = require("path");
const luaparse = require("luaparse");
let ok = 0, bad = 0;
function walk(dir) {
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) walk(p);
    else if (p.endsWith(".lua")) {
      try { luaparse.parse(fs.readFileSync(p, "utf8"), { luaVersion: "5.1" }); ok++; }
      catch (err) { bad++; console.log("FAIL", p, err.message); }
    }
  }
}
for (const d of process.argv.slice(2)) walk(d);
console.log(`${ok}/${ok + bad} Lua files parse as Lua 5.1`);
process.exit(bad ? 1 : 0);

// Builds docs\reports\phase5\S8-smoke\smoke.txt: the commands, their output, and the smoke kit's
// source verbatim, with the user folder written as %USERPROFILE% and the scratch folder as <scratch>.
// Every run has an expected outcome (pass, or for a mutant: fail exactly the checks named). If any
// run differs, the transcript still gets written, headed "BUILD FAILED", and this script exits 1
// (Greptile, PR #40 rounds 2 and 3).
import { execFileSync } from "node:child_process";
import { cpSync, mkdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import os from "node:os";
import path from "node:path";

function redact(text: string): string {
  let t = text;
  for (const [from, to] of [[kit, "<scratch>"], [os.homedir(), "%USERPROFILE%"]] as const) {
    for (const form of [from, from.replaceAll("\\", "\\\\"), from.replaceAll("\\", "/")]) {
      t = t.replace(new RegExp(form.replace(/[.*+?^${}()|[\]\\]/g, "\\$&"), "gi"), to);
    }
  }
  return t;
}

/**
 * Runs one command. `expect` "pass": it must exit 0. `expect` a list of check names: it must exit
 * non-zero with exactly those "FAIL <name>" lines and no Lua or task error, so a mutant that fails
 * for another reason (it does not load, another check breaks) is not taken as caught (Greptile,
 * PR #40 round 3).
 */
function run(label: string, args: string[], expect: "pass" | readonly string[]): string {
  let text: string;
  let ok = true;
  try {
    text = execFileSync("node", args, { cwd: repo, encoding: "utf8" });
  } catch (err) {
    ok = false;
    text = String((err as { stdout?: string }).stdout ?? "") + "\nEXIT NON-ZERO";
  }
  let as: string;
  if (expect === "pass") {
    as = ok ? "as expected" : "UNEXPECTED";
  } else {
    const failed = text.split("\n").filter((l) => l.startsWith("FAIL ")).map((l) => l.slice(5).trim()).sort();
    const wanted = [...expect].sort();
    const errored = /LUA ERROR|TASK ERROR/.test(text);
    const same = failed.length === wanted.length && failed.every((f, i) => f === wanted[i]);
    as = !ok && same && !errored ? "as expected" : `UNEXPECTED (failed checks: ${JSON.stringify(failed)}${errored ? ", with a Lua or task error" : ""})`;
  }
  if (as !== "as expected") unexpected.push(label);
  const wantText = expect === "pass" ? "pass" : `fail exactly ${JSON.stringify(expect)}`;
  return `\n=== ${label} (expected: ${wantText}; ${ok ? "passed" : "failed"}, ${as})\n> node ${args.join(" ")}\n${text.trimEnd()}\n`;
}
