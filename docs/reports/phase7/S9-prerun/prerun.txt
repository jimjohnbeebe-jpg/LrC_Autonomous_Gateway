--- Appendix C: zz-probe.ts ---
// Claude Code's probe: DPI as each side sees it, and CPU for static vs pulsing, visible vs hidden.
import koffi from "koffi";
import { execSync, spawn } from "node:child_process";
import { hudLog, hudPids, killTree, pollUntil, sleep, startStub } from "./harness.ts";
import { cpuMs, deckRect, dpi, mainWindow, pidsByExe, privateBytes, rect, tree, visible } from "./win32.ts";

const user32 = koffi.load("user32.dll");
const GetWindowDpiAwarenessContext = user32.func("intptr_t __stdcall GetWindowDpiAwarenessContext(uintptr_t h)");
const GetAwarenessFromDpiAwarenessContext = user32.func("int __stdcall GetAwarenessFromDpiAwarenessContext(intptr_t c)");
const aw = (h: number): number => GetAwarenessFromDpiAwarenessContext(GetWindowDpiAwarenessContext(h)) as number;

process.env.LRC_AVG_S9_LR_EXE = "notepad.exe";
const before = new Set(pidsByExe("notepad.exe"));
spawn("notepad.exe", [], { detached: true, stdio: "ignore" }).unref();
await sleep(2500);
const np = pidsByExe("notepad.exe").find((p) => !before.has(p) && mainWindow(p)) ?? 0;
const npH = mainWindow(np);
const stub = await startStub();
stub.cmd({ cmd: "spawn" });
const sp = await stub.waitFor((e) => e.ev === "spawned", 5000);
const pid = Number(sp?.pid);
await stub.waitFor((e) => e.ev === "paint", 15000);
const hwnd = Number(hudLog(pid).find((e) => e.ev === "start")?.hwnd);
await sleep(1000);
console.log("notepad dpi", dpi(npH), "awareness", aw(npH), "| hud dpi", dpi(hwnd), "awareness", aw(hwnd));
console.log("hud rect", rect(hwnd), "expected", deckRect(npH));
const cpu = async (label: string): Promise<void> => {
  const t = tree(pid);
  const c0 = t.map(cpuMs);
  const w0 = performance.now();
  await sleep(15000);
  const dt = performance.now() - w0;
  const per = t.map((p, i) => ({ p, pct: (((cpuMs(p) ?? 0) - (c0[i] ?? 0)) / dt) * 100 }));
  const mem = t.reduce((a, p) => a + (privateBytes(p) ?? 0), 0) / 2 ** 20;
  console.log(label, "total %", per.reduce((a, x) => a + x.pct, 0).toFixed(2), "per", per.map((x) => x.pct.toFixed(2)).join(" "), "mem MiB", mem.toFixed(1), "visible", visible(hwnd));
};
stub.cmd({ cmd: "state", name: "C" });
await cpu("visible static (C)");
stub.cmd({ cmd: "state", name: "W" });
await cpu("visible pulsing (W)");
stub.cmd({ cmd: "spike", action: "hide" });
await pollUntil(() => !visible(hwnd), 2000);
await cpu("hidden pulsing (W)");
stub.cmd({ cmd: "state", name: "W" });
stub.cmd({ cmd: "spike", action: "hide" });
stub.cmd({ cmd: "state", name: "C" });
await sleep(300);
stub.cmd({ cmd: "spike", action: "hide" });
await cpu("hidden static (C)");
stub.cmd({ cmd: "quit" });
for (const p of hudPids()) await killTree(tree(p));
execSync(`taskkill /PID ${np} /F`);
process.exit(0);

--- Appendix E: zz-follow.ts ---
import koffi from "koffi";
import { execSync, spawn } from "node:child_process";
import { hudLog, hudPids, killTree, now, pollUntil, sleep, startStub } from "./harness.ts";
import { deckRect, pidsByExe, rect, tree, visible, mainWindow, topmost } from "./win32.ts";
const u = koffi.load("user32.dll");
const ShowWindow = u.func("bool __stdcall ShowWindow(uintptr_t h, int cmd)");
const SetWindowPos = u.func("bool __stdcall SetWindowPos(uintptr_t h, uintptr_t after, int x, int y, int cx, int cy, uint32_t f)");
process.env.LRC_AVG_S9_LR_EXE = "notepad.exe";
spawn("notepad.exe", [], { detached: true, stdio: "ignore" }).unref();
await sleep(2500);
const np = pidsByExe("notepad.exe")[0] ?? 0; const npH = mainWindow(np);
const stub = await startStub();
stub.cmd({ cmd: "spawn" });
const pid = Number((await stub.waitFor((e) => e.ev === "spawned", 5000))?.pid);
await stub.waitFor((e) => e.ev === "paint", 15000);
await sleep(800);
const hud = Number(hudLog(pid).find((e) => e.ev === "start")?.hwnd);
const eq = (a: any, b: any) => JSON.stringify(a) === JSON.stringify(b);
console.log("start in place", eq(rect(hud), deckRect(npH)), JSON.stringify(rect(hud)), JSON.stringify(deckRect(npH)), "topmost", topmost(hud));
const r0 = rect(npH);
for (const dx of [150, -150]) {
  const t = now(); SetWindowPos(npH, 0, r0.left + dx, r0.top + 40, r0.right - r0.left - 100, r0.bottom - r0.top - 60, 0x0014 /* NOZORDER|NOACTIVATE */);
  const ok = await pollUntil(() => eq(rect(hud), deckRect(npH)), 1000, 2);
  console.log("move", dx, "in place ms", ok === null ? null : (ok - t).toFixed(1));
}
let t = now(); ShowWindow(npH, 6 /* SW_MINIMIZE */);
let ok = await pollUntil(() => !visible(hud), 1000, 2); console.log("minimise -> hidden ms", ok === null ? null : (ok - t).toFixed(1));
await sleep(500);
t = now(); ShowWindow(npH, 9 /* SW_RESTORE */);
ok = await pollUntil(() => visible(hud) && eq(rect(hud), deckRect(npH)), 1000, 2); console.log("restore -> shown in place ms", ok === null ? null : (ok - t).toFixed(1));
const log = hudLog(pid); console.log("log counts", Object.entries(log.reduce((a: any, e) => ((a[e.ev] = (a[e.ev] ?? 0) + 1), a), {})));
stub.cmd({ cmd: "quit" });
for (const p of hudPids()) await killTree(tree(p));
execSync(`taskkill /PID ${np} /F`);
process.exit(0);
