== Contract (HudState.check against the engine's hudUpdatePayloadSchema) ==
--  minimal: plugin takes; engine schema takes
--  every field: plugin takes; engine schema takes
--  an unknown field: plugin refuses (hud_update.colour is not a known field); engine schema refuses
--  an unknown target field: plugin refuses (hud_update.target.camera is not a known field); engine schema refuses
--  an unknown stage: plugin refuses (hud_update.stage must be one of begin, pass0, applying, acquiring_preview, metrics, awaiting_claude, awaiting_pick, awaiting_approval, converged, target_changed, accepted, aborted, ended); engine schema refuses
--  no target: plugin refuses (hud_update.target is required); engine schema refuses
--  no session id: plugin refuses (hud_update.session_id is required); engine schema refuses
--  seq 0: plugin refuses (hud_update.seq is out of range); engine schema refuses
--  seq 1.5: plugin refuses (hud_update.seq must be a whole number); engine schema refuses
--  seq as text: plugin refuses (hud_update.seq must be a number); engine schema refuses
--  session id of 64 bytes (32 x e-acute): plugin takes; engine schema takes
--  session id of 66 bytes (33 x e-acute): plugin refuses (hud_update.session_id must be a non-empty string of at most 64 bytes); engine schema refuses
--  an empty session id: plugin refuses (hud_update.session_id must be a non-empty string of at most 64 bytes); engine schema refuses
--  variant D: plugin refuses (hud_update.variants[2] must be one of A, B, C); engine schema refuses
--  12 delta rows: plugin takes; engine schema takes
--  13 delta rows: plugin refuses (hud_update.deltas has more than 12 entries); engine schema refuses
--  a delta row without its slider: plugin refuses (hud_update.deltas[1].slider is required); engine schema refuses
--  guardrail amber: plugin refuses (hud_update.guardrail.status must be one of green, clamped, refused, corrected, unmet, undone); engine schema refuses
--  a guardrail without its status: plugin refuses (hud_update.guardrail.status is required); engine schema refuses
--  open as text: plugin refuses (hud_update.open must be true or false); engine schema refuses
--  note as an object: plugin refuses (hud_update.note must be text); engine schema refuses
--  pass 99: plugin takes; engine schema takes
--  pass 100: plugin refuses (hud_update.pass is out of range); engine schema refuses
--  max_passes 0: plugin refuses (hud_update.max_passes is out of range); engine schema refuses
--  settings with log_dir: plugin refuses (hud_update.settings.log_dir is not a known field); engine schema refuses
--  decay with 9 values: plugin refuses (hud_update.settings.decay has more than 8 entries); engine schema refuses
--  17 session photos: plugin refuses (hud_update.session_photos has more than 16 entries); engine schema refuses
--  session photos as an object: plugin refuses (hud_update.session_photos must be a list); engine schema refuses
--  an empty answered_click_id: plugin refuses (hud_update.answered_click_id must be a non-empty string of at most 64 bytes); engine schema refuses
--  a 200-byte file name: plugin takes; engine schema takes
--  a 200-byte UTF-8 file name (100 x e-acute): plugin takes; engine schema takes
--  numbers where text is shown: plugin takes; engine schema takes
--  the target as a list: plugin refuses (hud_update.target.1 is not a known field); engine schema refuses
--  the payload as a list: plugin refuses (hud_update.1 is not a known field); engine schema refuses
--  an empty object where a list belongs: plugin takes; engine schema refuses

-- In-memory files (fengari has no io.open).
io = io or {}
io.open = function(path, mode)
    mode = mode or "r"
    if mode:sub(1, 1) == "r" then
        if F.files[path] == nil then return nil, path .. ": No such file" end
        local text = F.files[path]
        return { read = function() return text end, close = function() end }
    end
    if mode:sub(1, 1) == "w" then F.files[path] = "" end
    if F.files[path] == nil then F.files[path] = "" end
    return {
        write = function(self, ...) for _, s in ipairs({ ... }) do F.files[path] = F.files[path] .. tostring(s) end end,
        close = function() end,
    }
end

-- A tiny cooperative scheduler: every task is a coroutine; sleep and yield hand control back.
-- PHASE5_PLAN row 4 changed the clock: one round is F.TICK (0.2 s) of virtual time, and a task that
-- sleeps s seconds is resumed in the first round at or after its wake time. Before, every sleeping
-- task moved the shared clock by its own step, so with several tasks asleep time ran several times
-- too fast and timing checks depended on the order tasks ran in. A task spawned during a round first
-- runs in the next one.
F.TICK = 0.2
function F.spawn(name, fn)
    F.tasks[#F.tasks + 1] = { name = name, co = coroutine.create(fn), wake = F.clock }
end
function F.run(rounds)
    for _ = 1, rounds or 1 do
        local current = F.tasks
        F.tasks = {}
        local live = {}
        for _, t in ipairs(current) do
            if coroutine.status(t.co) == "suspended" and t.wake <= F.clock + 1e-9 then
                F.running = t
                local ok, err = coroutine.resume(t.co)
                F.running = nil
                if not ok then F.errors[#F.errors + 1] = t.name .. ": " .. tostring(err) end
            end
            if coroutine.status(t.co) ~= "dead" then live[#live + 1] = t end
        end
        for _, t in ipairs(F.tasks) do live[#live + 1] = t end
        F.tasks = live
        F.clock = F.clock + F.TICK
    end
end
-- Run fn in a task until it ends; returns what it returned.
function F.runTask(fn)
    local out
    F.spawn("runTask", function() out = table.pack(fn()) end)
    for _ = 1, 1000 do
        if #F.tasks == 0 then break end
        F.run(1)
    end
    if not out then return nil end
    return table.unpack(out, 1, out.n)
end

-- (row 3's fake, unchanged). Modelled on S8's smoke fake (docs\reports\phase5\S8-smoke\smoke.txt,

--   - LrView.osFactory records every view it builds; push_buttons are kept so a scenario can click
--     them by title (a bound title is read from the window's property table);
--   - a click on a button whose bound `enabled` is not true is not delivered (S8: greyed clicks 0);
--   - a click runs the button's action on the main Lua thread, where nothing can yield (S8: the

local function titleOf(w, b)
    local t = b.spec.title
    if type(t) == "table" then return w.props[t.bound] end
    return t
end

-- "clicked", "disabled", or "no window"; the action runs on the main thread.
function FH.click(title)
    local w = FH.current
    if not (w and w.open) then return "no window" end
    for _, b in ipairs(w.buttons) do
        if titleOf(w, b) == title then
            local en = b.spec.enabled
            if type(en) == "table" and w.props[en.bound] ~= true then return "disabled" end
            b.spec.action()
            return "clicked"
        end
    end
    error("no button titled " .. tostring(title))
end

-- Each button's title and whether a click would be delivered, in the order the window lists them.
function FH.buttons()
    local w = FH.current
    local out = {}
    if not (w and w.open) then return out end
    for _, b in ipairs(w.buttons) do
        local en = b.spec.enabled
        out[#out + 1] = { title = titleOf(w, b), enabled = type(en) ~= "table" or w.props[en.bound] == true }
    end
    return out
end

end

function runLua(pluginDir: string, script: string): any {
  const L = lauxlib.luaL_newstate();
  lualib.luaL_openlibs(L);
  const dirs = pluginDir.split(";").map((d) => `${d}/?.lua`).join(";");
  const setup = `package.path = [[${dirs};${slash(here)}/?.lua]]\ndofile([[${slash(here)}/fake-lr.lua]])\ndofile([[${slash(here)}/fake-hud.lua]])\nCASES = [==[${casesJson}]==]`;
  if (lauxlib.luaL_dostring(L, to_luastring(setup)) !== lua.LUA_OK) throw new Error(lua.lua_tojsstring(L, -1));
  if (lauxlib.luaL_dofile(L, to_luastring(slash(path.join(here, script)))) !== lua.LUA_OK) throw new Error(`${script}: ${lua.lua_tojsstring(L, -1)}`);
  return JSON.parse(lua.lua_tojsstring(L, -1));
}

type Check = { name: string; ok: boolean };
const SELECTION_OK = "Selection: the session's photo.";
function checksOf(run: any): Check[] {

  // read back as whole e-acutes.

  check(r2.applied && r2.shown && !r2.opened && second.windows === 1 && second.open === 1, "H2 open once: a second open and Show HUD while it is open open no second window");

    "H28 a menu item used with the HUD closed opens it and shows the outcome there (row 4 probe)");
  return checks;
}

let failures = 0;
function report(checks: Check[]): void {
  for (const c of checks) {
    console.log(`${c.ok ? "PASS" : "FAIL"}  ${c.name}`);
    if (!c.ok) failures++;
  }
}

// Mutants: each must fail exactly its named checks.
function mutant(file: string, from: string, to: string, expect: string[]): void {
  const source = readFileSync(path.join(newDir, file), "utf8").replace(/\r\n/g, "\n"); // the working tree may have CRLF
  const changed = source.replace(from, to);
  const dir = path.join(here, "mutant");
  rmSync(dir, { recursive: true, force: true });
  mkdirSync(dir, { recursive: true });
  writeFileSync(path.join(dir, file), changed);
  let failed: string[];
  try {
    failed = checksOf(runLua(`${slash(dir)};${newDir}`, "hud-cases.lua")).filter((c) => !c.ok).map((c) => c.name.split(" ")[0] as string);
  } catch (e) {
    failed = [`raised: ${(e as Error).message.slice(0, 120)}`];
  }
  const got = [...new Set(failed)].sort();
  const ok = changed !== source && JSON.stringify(got) === JSON.stringify([...expect].sort());
  console.log(`${ok ? "PASS" : "FAIL"}  mutant ${file} (${to.trim().slice(0, 80)}): fails ${JSON.stringify(got)}, expected ${JSON.stringify(expect)}`);
  if (!ok) failures++;
}
